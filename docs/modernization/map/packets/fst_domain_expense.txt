/**
 * 差旅申请聚合根。
 *
 * <p>业务（Why）：聚合只表达差旅业务状态机，不接入 HTTP 或 Flowable 编排；工作流 Hook 只能通过
 * 已冻结的 transition 将可靠流程结果镜像为业务状态。
 */
public class TravelApplication {

    private static final Pattern DOCUMENT_NUMBER_PATTERN = Pattern.compile("TR-\\d{4}-\\d{6}");

    private Long id;
    private Long version;
    private final Long applicantId;
    private final EmployeeSnapshot applicantSnapshot;
    private final LocalDate applicationDate;
    private String travelType;
    private final String documentNumber;
    private LocalDateTime planStartTime;
    private LocalDateTime planEndTime;
    private List<TravelApplicationLine> lines;
    private TravelApplicationStatus status;
    /** 费用归集确认写入主表的只读派生金额，不由聚合保存路径回写。 */
    private BigDecimal payableAmount;

    private TravelApplication(Long applicantId, EmployeeSnapshot applicantSnapshot, LocalDate applicationDate,
                              String travelType, String documentNumber,
                              LocalDateTime planStartTime, LocalDateTime planEndTime,
                              List<TravelApplicationLine> lines, TravelApplicationStatus status) {
        this.applicantId = applicantId;
        this.applicantSnapshot = applicantSnapshot;
        this.applicationDate = applicationDate;
        this.travelType = travelType;
        this.documentNumber = documentNumber;
        this.planStartTime = planStartTime;
        this.planEndTime = planEndTime;
        this.lines = lines;
        this.status = status;
        this.version = 0L;
    }

    public static TravelApplication createDraft(Long applicantId, String travelType, String documentNumber,
                                                List<TravelApplicationLine> lines,
                                                LocalDateTime ignoredPlanStartTime,
                                                LocalDateTime ignoredPlanEndTime,
                                                Clock clock) {
        return createDraft(applicantId, null, travelType, documentNumber, lines,
                ignoredPlanStartTime, ignoredPlanEndTime, clock);
    }

    public static TravelApplication createDraft(Long applicantId, EmployeeSnapshot applicantSnapshot,
                                                String travelType, String documentNumber,
                                                List<TravelApplicationLine> lines,
                                                LocalDateTime ignoredPlanStartTime,
                                                LocalDateTime ignoredPlanEndTime,
                                                Clock clock) {
        if (applicantId == null) {
            throw new IllegalArgumentException("申请人不能为空");
        }
        if (travelType == null || travelType.trim().isEmpty()) {
            throw new IllegalArgumentException("差旅类型不能为空");
        }
        if (documentNumber == null || documentNumber.trim().isEmpty()) {
            throw new IllegalArgumentException("差旅单号不能为空");
        }
        if (!DOCUMENT_NUMBER_PATTERN.matcher(documentNumber).matches()) {
            throw new IllegalArgumentException("差旅单号格式不合法");
        }
        if (lines == null || lines.isEmpty()) {
            throw new IllegalArgumentException("B区明细行不能为空");
        }
        if (clock == null) {
            throw new IllegalArgumentException("业务时钟不能为空");
        }

        List<TravelApplicationLine> orderedLines = new ArrayList<>(lines);
        Collections.sort(orderedLines, Comparator.comparingInt(TravelApplicationLine::getSequence));
        List<TravelApplicationLine> orderedActiveLines = activeLines(orderedLines);
        if (orderedActiveLines.isEmpty()) {
            throw new IllegalArgumentException("B区明细行不能为空");
        }
        validateLineSequences(orderedActiveLines);
        validateLineContinuity(orderedActiveLines);

        LocalDateTime minLineStartTime = orderedActiveLines.get(0).getStartTime();
        LocalDateTime maxLineEndTime = orderedActiveLines.get(orderedActiveLines.size() - 1).getEndTime();
        LocalDateTime planStartTime = minLineStartTime;
        LocalDateTime planEndTime = maxLineEndTime;
        validatePlanBounds(planStartTime, planEndTime, minLineStartTime, maxLineEndTime);

        return new TravelApplication(applicantId, applicantSnapshot, LocalDate.now(clock),
                travelType, documentNumber, planStartTime, planEndTime,
                orderedLines, TravelApplicationStatus.DRAFT);
    }

    public static TravelApplication restore(Long id, Long version, Long applicantId, String travelType, String documentNumber,
                                            LocalDateTime planStartTime, LocalDateTime planEndTime,
                                            List<TravelApplicationLine> lines,
                                            TravelApplicationStatus status) {
        return restore(id, version, applicantId, null, null, travelType, documentNumber,
                planStartTime, planEndTime, lines, status);
    }

    public static TravelApplication restore(Long id, Long version, Long applicantId,
                                            EmployeeSnapshot applicantSnapshot, LocalDate applicationDate,
                                            String travelType, String documentNumber,
                                            LocalDateTime planStartTime, LocalDateTime planEndTime,
                                            List<TravelApplicationLine> lines,
                                            TravelApplicationStatus status) {
        return restore(id, version, applicantId, applicantSnapshot, applicationDate, travelType, documentNumber,
                planStartTime, planEndTime, lines, status, null);
    }

    public static TravelApplication restore(Long id, Long version, Long applicantId,
                                            EmployeeSnapshot applicantSnapshot, LocalDate applicationDate,
                                            String travelType, String documentNumber,
                                            LocalDateTime planStartTime, LocalDateTime planEndTime,
                                            List<TravelApplicationLine> lines,
                                            TravelApplicationStatus status, BigDecimal payableAmount) {
        if (id == null || id <= 0) {
            throw new IllegalArgumentException("差旅申请ID不能为空");
        }
        if (status == null) {
            throw new IllegalArgumentException("差旅申请状态不能为空");
        }
        TravelApplication draft = createDraft(applicantId, travelType, documentNumber, lines, planStartTime,
                planEndTime, Clock.fixed(planStartTime.atZone(java.time.ZoneId.systemDefault()).toInstant(),
                        java.time.ZoneId.systemDefault()));
        TravelApplication application = new TravelApplication(applicantId, applicantSnapshot, applicationDate,
                travelType, documentNumber, draft.planStartTime, draft.planEndTime,
                draft.lines, draft.status);
        application.id = id;
        application.version = version == null ? 0L : version;
        application.status = status;
        application.payableAmount = payableAmount;
        return application;
    }

    public TravelApplication copy() {
        List<TravelApplicationLine> copiedLines = new ArrayList<>();
        for (TravelApplicationLine line : this.lines) {
            copiedLines.add(line.copy());
        }
        TravelApplication copied = new TravelApplication(this.applicantId, this.applicantSnapshot,
                this.applicationDate, this.travelType, this.documentNumber,
                this.planStartTime, this.planEndTime, copiedLines, this.status);
        copied.id = this.id;
        copied.version = this.version;
