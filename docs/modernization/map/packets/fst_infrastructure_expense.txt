    public TravelPlatformOrgSynchronizationRepositoryImpl(
            TravelPlatformOrgSynchronizationMapper synchronizationMapper,
            IdentifierGenerator identifierGenerator,
            PlatformTransactionManager transactionManager,
            DataSource dataSource,
            Clock clock) {
        if (synchronizationMapper == null || identifierGenerator == null
                || transactionManager == null || dataSource == null || clock == null) {
            throw new IllegalArgumentException("组织同步 Repository 依赖不能为空");
        }
        this.synchronizationMapper = synchronizationMapper;
        this.identifierGenerator = identifierGenerator;
        this.transactionManager = transactionManager;
        this.dataSource = dataSource;
        this.jdbcTemplate = new JdbcTemplate(dataSource);
        this.clock = clock;
    }

    @Override
    public Optional<TravelPlatformOrgSynchronizationRun> findLatestRun(String targetCompanyKey) {
        this.requireTargetCompanyKey(targetCompanyKey);
        TravelPlatformOrgSynchronizationRunDO source =
                synchronizationMapper.selectLatestRun(targetCompanyKey);
        return source == null
                ? Optional.empty()
                : Optional.of(TravelPlatformOrgSynchronizationMapping.toRun(source));
    }

    @Override
    public Optional<Instant> findLatestRemoteVerificationTime(String targetCompanyKey) {
        this.requireTargetCompanyKey(targetCompanyKey);
        LocalDateTime value = synchronizationMapper.selectLatestRemoteVerificationTime(targetCompanyKey);
        return value == null ? Optional.empty() : Optional.of(value.toInstant(ZoneOffset.UTC));
    }

    @Override
    public long allocateFenceToken(String targetCompanyKey) {
        this.requireTargetCompanyKey(targetCompanyKey);
        Long fenceToken = synchronizationMapper.selectNextFenceToken();
        if (fenceToken == null || fenceToken <= 0) {
            throw new IllegalStateException("数据库未返回有效组织同步 fenceToken");
        }
        return fenceToken;
    }

    @Override
    public void claimFence(String targetCompanyKey, long fenceToken) {
        this.requireTargetCompanyKey(targetCompanyKey);
        this.requirePositive("fenceToken", fenceToken);
        // fence claim 是后续 provider I/O 与数据库写入的线性化边界。
        this.executeIndependentWrite("组织同步 fence claim", () -> {
            synchronizationMapper.claimFence(targetCompanyKey, fenceToken);
            return null;
        });
    }

    @Override
    public void recoverOrphanRuns(String targetCompanyKey, long fenceToken) {
        this.requireTargetCompanyKey(targetCompanyKey);
        this.requirePositive("fenceToken", fenceToken);
        // 新 owner 在建立运行证据前收敛同一目标企业遗留 running。
        this.executeIndependentWrite("组织同步 orphan recovery", () -> {
            synchronizationMapper.recoverOrphanRuns(targetCompanyKey, fenceToken);
            return null;
        });
    }
