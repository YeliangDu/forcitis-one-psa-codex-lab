): TravelApplicationPayload {
  return {
    expectedVersion: application.version ?? null,
    travelType: application.travelType,
    explicitPlanStartTime: application.explicitPlanStartTime

function normalizeApplication(data: TravelApplication): TravelApplication {
  return {
    ...data,
    version: data.version ?? null,
    explicitPlanStartTime: toDateTimeInput(data.explicitPlanStartTime || data.planStartTime),
    explicitPlanEndTime: toDateTimeInput(data.explicitPlanEndTime || data.planEndTime),
    lines: (data.lines || []).map((line, index) => {
      const snapshot = parseBusinessObjectSnapshot(line.businessObjectSnapshot)
      return {
        ...line,
        sequence: line.sequence || index + 1,
        businessObjectCode: line.businessObjectCode || String(snapshot.businessObjectCode || ''),
        businessObjectName: line.businessObjectName || String(snapshot.businessObjectName || ''),
        costCenterId: line.costCenterId ?? (typeof snapshot.costCenterId === 'string' || typeof snapshot.costCenterId === 'number'
          ? snapshot.costCenterId
          : null),
        costCenterCode: line.costCenterCode || String(snapshot.costCenterCode || ''),
        costCenterName: line.costCenterName || String(snapshot.costCenterName || ''),
        startTime: toDateTimeInput(line.startTime),
        endTime: toDateTimeInput(line.endTime),
        purpose: line.purpose || '',
        secondaryAllocationType: line.secondaryAllocationType || String(snapshot.secondaryAllocationType || SECONDARY_ALLOCATION_TYPE.NONE),
        secondaryAllocationReferencedId: line.secondaryAllocationReferencedId ?? (typeof snapshot.secondaryAllocationReferencedId === 'string' || typeof snapshot.secondaryAllocationReferencedId === 'number'
          ? snapshot.secondaryAllocationReferencedId
          : null),
        secondaryAllocationReferencedCode: line.secondaryAllocationReferencedCode || String(snapshot.secondaryAllocationReferencedCode || ''),
        secondaryAllocationLabel: line.secondaryAllocationLabel || String(snapshot.secondaryAllocationLabel || ''),
        rowStatus: line.rowStatus || ROW_APPROVAL_STATUS.DRAFT_ROW
      }
    })
  }
}

function lineValidationMessage(line: TravelApplicationLine, now: number | null = toTimestamp(new Date())): string {
  if (!line.fromCityId || !line.toCityId) return '请选择出发地和目的地'
  if (!line.businessObjectId || !line.businessObjectType) return '请选择业务对象'
  if (!line.startTime || !line.endTime) return '请选择开始和结束时间'
  const startTime = toTimestamp(line.startTime)
  const endTime = toTimestamp(line.endTime)
  if (startTime === null || endTime === null) return '时间格式不正确'
  if (now !== null && startTime < now) return '开始时间不能早于当前时间'
  if (endTime <= startTime) return '结束时间必须晚于开始时间'
  if (!String(line.purpose || '').trim()) return '请填写出差事由'
  return ''
}
