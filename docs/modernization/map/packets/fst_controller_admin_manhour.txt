/**
 * psa_time_sheet_works 历史脏数据修复逻辑（按错误类型分方法：fixTypeA / fixTypeF ...）。
 * <p>
 * 修复对象是 psa_time_sheet_works_error 表里的行（先在错误表订正、按 eId 更新，验证无误后再回写主表）。
 * HTTP 执行入口见 {@code FixController}。
 */
@Slf4j
@Service
public class FixPsaTimeSheetWorks {

    /** 审批通过状态值（工时子表：3） */
    private static final Integer STATUS_PASSED = CommonConstants.ApproveStatus.通过.getValue();
    /** 流程实例审批结果：通过（bpm_process_instance_ext.result = 2） */
    private static final Integer PROCESS_RESULT_PASSED = CommonConstants.BpmProcessInstanceResult.通过;
    /** 修复流水结果值 */
    private static final String RESULT_SUCCESS = "成功";
    private static final String RESULT_FAIL = "失败";

    @Resource
    private TimeSheetRateService rateService;
    @Resource
    private EmployeeArchiveCacheService employeeArchiveCache;
    @Resource
    private TempSystemUserMapper tempSystemUserMapper;
    @Resource
    private TempSystemDeptMapper tempSystemDeptMapper;
    @Resource
    private ManHourDirtyDetector dirtyDetector;
    @Resource
    private ErrorManHourItemMapper errorManHourItemMapper;
    @Resource
    private ErrManHourItemLogMapper errManHourItemLogMapper;
    @Resource
    private ManHourApproveStatusRevisionMapper revisionMapper;
    @Resource
    private ManHourMapper manHourMapper;
    @Resource
    private ManHourItemMapper manHourItemMapper;
    @Resource
    private CustomizedBpmOAMapper customizedBpmOAMapper;
    @Resource
    private BpmProcessInstanceExtMapper processInstanceExtMapper;

    /**
     * 修复 A 类（费率九件套缺失）。
     * <p>
     * 输入：{@code creator}（用户ID）+ {@code worksDate} → 调 {@link TimeSheetRateService#resolveRates}
     * 还原当时职级对应的费率；九件套<b>直接映射</b>（对齐 {@code ManHourServiceImpl.applySubmitRateFields}，
     * 无二次计算，金额留待后续 fix）。按 {@code eId} 更新错误表。
     * <ul>
     *   <li>成功：写入九件套，{@code errProcessType} 累加 {@code A}；</li>
     *   <li>失败（resolveRates 抛 503：档案缺失/费率未命中/多命中）：把 message 记进 {@code errProcessMsg}。</li>
     * </ul>
     *
     * @return true=修复成功；false=失败（已记录 errProcessMsg，不中断批量）
     */
    public boolean fixTypeA(ErrManHourItemDO err) {
        // creator 即工时归属人 userId；worksDate 决定"当时"职级与费率
        Long userId = parseUserId(err.getCreator());
        LocalDate worksDate = err.getWorksDate() == null ? null : err.getWorksDate().toLocalDate();
        if (userId == null || worksDate == null) {
            logFix(err, "A", RESULT_FAIL, "creator 或 worksDate 无效：creator=" + err.getCreator()
                    + "，worksDate=" + err.getWorksDate());
            return false;
        }
        // hours 仅影响金额；九件套不依赖它。传 approvalHours（无则 ZERO），对齐提交口径
        BigDecimal hours = err.getApprovalHours() != null ? err.getApprovalHours() : BigDecimal.ZERO;
        try {
            RateResult rate = rateService.resolveRates(userId, worksDate, hours);
            // 九件套直接塞（对齐 applySubmitRateFields:1250-1258，无额外逻辑计算）
            err.setStandardRateId(rate.getStandardRateId());
            err.setStandardRateDetailId(rate.getStandardRateDetailId());
            err.setStandardCostRate(rate.getStandardCostRate());
            err.setStandardCostUnit(rate.getStandardCostUnit());
            err.setBillRateId(rate.getBillRateId());
            err.setBillRateDetailId(rate.getBillRateDetailId());
            err.setBillRate(rate.getBillRate());
            err.setBillRateUnit(rate.getBillRateUnit());
            err.setBillableFlag(rate.getBillableFlag());
            // 当前态：errProcessType 累加 A + 落九件套
            err.setErrProcessType(appendProcessType(err.getErrProcessType(), "A"));
            errorManHourItemMapper.updateById(err); // 按 eId 更新
            // 修复流水：记取值来源
            logFix(err, "A", RESULT_SUCCESS, "费率九件套来源 resolveRates 命中（userId=" + userId
                    + "，worksDate=" + worksDate + "，standardCostRate=" + rate.getStandardCostRate()
                    + "，billRate=" + rate.getBillRate() + "）");
            return true;
        } catch (ServiceException ex) {
            // resolveRates 严格化后的 503：档案缺失 / 费率未命中 / 多命中
            logFix(err, "A", RESULT_FAIL, ex.getMessage());
            return false;
        } catch (Exception ex) {
            // 兜底：单行异常不拖垮批量
            logFix(err, "A", RESULT_FAIL, "异常：" + ex.getMessage());
            return false;
        }
    }

    /**
     * 修复 F 类（审批通过但 approved_time 缺失）。审批时间不可重算，按<b>三级兜底</b>找回真实审批时刻：
     * <ol>
     *   <li><b>留痕表</b>：{@code psa_time_sheet_approve_status_revision} 中该 tsheet 最近一次
     *       {@code to_approve_status=通过} 的 {@code create_time}；</li>
     *   <li><b>流程引擎</b>：tsheet → {@code bpm_oa_customized.params_id} → {@code process_instance_id}
     *       → {@code bpm_process_instance_ext}（{@code result=通过}）的最新 {@code end_time}；</li>
     *   <li><b>提交时间兜底</b>：前两者都无（提交即通过、无审批流的老数据）时，取本明细行原始提交时间
     *       {@code psa_time_sheet_works.create_time}（按原始明细行 id，提交时刻≈审批时刻）。</li>
     * </ol>
     * 非留痕表来源（②③）在 {@code errProcessMsg} 留痕，便于区分数据来源与保真度。
     *
     * @return true=修复成功；false=失败（已记录 errProcessMsg）
     */
    public boolean fixTypeF(ErrManHourItemDO err) {
        Long tsheetId = err.getTsheetId();
        if (tsheetId == null) {
            logFix(err, "F", RESULT_FAIL, "tsheetId 为空");
            return false;
        }
        // ① 留痕表优先
        LocalDateTime approvedTime = pickApprovedTimeFromRevision(tsheetId);
        String source = "留痕表 create_time";
        // ② 流程引擎 end_time
        if (approvedTime == null) {
            approvedTime = pickApprovedTimeFromProcessEngine(tsheetId);
            if (approvedTime != null) {
                source = "流程引擎 end_time（留痕表无「通过」记录）";
            }
        }
        // ③ 提交时间兜底（提交即通过）：留痕表/流程引擎均无审批记录，取本明细行原始提交时间 psa_time_sheet_works.create_time
        //    （提交即通过的老数据，提交时刻≈审批时刻；比 update_time 更贴近、且不被后续修改污染）
        if (approvedTime == null) {
            approvedTime = errorManHourItemMapper.selectCreateTimeById(err.getId());
            if (approvedTime != null) {
                source = "提交时间 psa_time_sheet_works.create_time（提交即通过，留痕表 / 流程引擎均无「通过」记录）";
            }
        }

        if (approvedTime == null) {
            logFix(err, "F", RESULT_FAIL, "留痕表 / 流程引擎 / 提交时间 均无法还原审批时间");
            return false;
        }

        err.setApprovedTime(approvedTime);
        err.setErrProcessType(appendProcessType(err.getErrProcessType(), "F"));
        errorManHourItemMapper.updateById(err); // 按 eId 更新
        logFix(err, "F", RESULT_SUCCESS, "approved_time=" + approvedTime + "，来源：" + source);
        return true;
