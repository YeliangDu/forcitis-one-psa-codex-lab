/**
 * 项目预算聚合根（WBS 口径：条款→交付包→任务包 + 项目级锁定，openspec add-project-budgeting）。
 * <p>不变式（聚合自证）：
 * <ul>
 *   <li>编制闸：wbs_budget_locked=1（审批通过锁定）或审批中（PENDING）后所有编制操作拒绝；
 *       保存预算=提交 PM_BUDGET 审批，通过才锁定（refs #723）</li>
 *   <li>导入幂等闸：已存在条款不可重复导入</li>
 *   <li>任务包：岗位必须在静态费率表取值域、预算工时非负（null 归一 0）、必须挂在本树交付包下</li>
 * </ul>
 * 汇总口径：预算成本=Σ任务包（工时×岗位成本费率）；预算毛利率按来源报价实际税率反算不含税收入，
 * 4 位小数，金额≤0 防除零置 null（与报价毛利率同口径）。
 * <p>违例抛 {@link BudgetRuleViolation}，由应用层翻译为带错误码的 ServiceException（B12）。
 */
public class ProjectBudget {

    /** 整单税率（不含税换算口径，与报价 DEFAULT_TAX_RATE 对齐） */
    private static final BigDecimal DEFAULT_TAX_RATE = new BigDecimal("0.06");

    /** 项目ID（psa_pm_project.id） */
    private final Long projectId;
    /** 项目编号（条款/交付包/任务包 code 派生前缀） */
    private final String projectCode;
    /** 关联商机ID（报价导入来源定位；可空） */
    private final Long opportunityId;
    /** 合同金额（psa_pm_project.amount，毛利率分母；可空） */
    private final BigDecimal contractAmount;
    /** 预算审批状态：审批中（编制/重复提交封锁） */
    public static final String APPROVAL_PENDING = "PENDING";
    /** 预算审批状态：已驳回（可修改后重新提交） */
    public static final String APPROVAL_REJECTED = "REJECTED";

    /** 预算锁定标记（wbs_budget_locked） */
    private boolean locked;
    /** 预算审批状态（wbs_budget_approval_status；null=从未提交或已通过） */
    private String approvalStatus;
    /** 最近一次 PM_BUDGET 流程实例号（wbs_budget_process_id；终态回写时绑定） */
    private String processInstanceId;
    /** 退回发起人放行标记（refs #1620，内存态不落库）：见 {@link #markReturnedToInitiator()} */
    private boolean returnedToInitiator;
    /** 条款树（仅聚合业务方法维护） */
    private final List<BudgetClause> clauses = new ArrayList<>();
    /** 费用类目金额（category→amount，refs #748；人力成本不在此，见 {@link BudgetCostCategory}） */
    private final java.util.Map<String, BigDecimal> costItems = new java.util.LinkedHashMap<>();
    /** 本次变更中被移除（档案已删且无已发生数据）的条款 ID，供仓储软删（refs #991） */
    private final List<Long> removedClauseIds = new ArrayList<>();
    /**
     * 预算费率取数城市（refs #1826，装载时由仓储解析：项目「预算所属城市」优先 → 预算来源报价城市 → null=不过滤）。
     * 只读快照，供详情回显与前端岗位费率下拉同城取数；不参与业务规则。
     */
    private String cityCategoryCode;

    private ProjectBudget(Long projectId, String projectCode, Long opportunityId,
                          BigDecimal contractAmount, boolean locked) {
        this.projectId = projectId;
        this.projectCode = projectCode;
        this.opportunityId = opportunityId;
        this.contractAmount = contractAmount;
        this.locked = locked;
    }

    /**
     * 仓储回装（无审批状态的便捷重载：从未提交审批的持久化态）。
     */
    public static ProjectBudget restore(Long projectId, String projectCode, Long opportunityId,
                                        BigDecimal contractAmount, boolean locked, List<BudgetClause> clauses) {
        return restore(projectId, projectCode, opportunityId, contractAmount, locked, null, null, clauses);
    }

    /**
     * 仓储回装（从持久化态重建，不触发业务校验；clauses 由仓储装载后填入）。
     */
    public static ProjectBudget restore(Long projectId, String projectCode, Long opportunityId,
                                        BigDecimal contractAmount, boolean locked, String approvalStatus,
                                        String processInstanceId, List<BudgetClause> clauses) {
        ProjectBudget budget = new ProjectBudget(projectId, projectCode, opportunityId, contractAmount, locked);
        budget.approvalStatus = approvalStatus;
        budget.processInstanceId = processInstanceId;
        if (clauses != null) {
            budget.clauses.addAll(clauses);
        }
        return budget;
    }

    /** 仓储回装费用类目（restore 后调用；不触发业务校验）。 */
    public void restoreCostItems(java.util.Map<String, BigDecimal> items) {
        if (items != null) {
            this.costItems.putAll(items);
        }
    }

    /** 仓储回装预算费率取数城市（refs #1826；restore 后调用，不触发业务校验）。 */
    public void restoreCityCategoryCode(String cityCategoryCode) {
        this.cityCategoryCode = cityCategoryCode;
    }

    public String getCityCategoryCode() {
        return cityCategoryCode;
    }

    // ===== 导入（初始化） =====

    /**
     * 初始化条款/交付包（首次装载，refs #976：条款层来自项目档案，交付层底稿来自赢单报价）：
     * 锁定闸 + 幂等闸；code 按 {项目编号}-C{序号} / {条款编号}D{序号} 派生——条款/交付包仅此一条
     * 创建路径且一期无删除，不存在软删占号问题；异常残留（如手工删库）撞号时由唯一约束兜底失败，不产生脏数据。
     */
    public void initializeClauses(List<ClauseImport> imports) {
        this.ensureEditable();
        if (!this.clauses.isEmpty()) {
            throw new BudgetRuleViolation(BudgetRuleViolation.Reason.ALREADY_INITIALIZED, "项目预算已存在条款，不可重复导入");
        }
        List<ClauseImport> safeImports = imports == null ? Collections.emptyList() : imports;
        int clauseSeq = 0;
        for (ClauseImport clauseImport : safeImports) {
            clauseSeq++;
            String clauseCode = this.projectCode + "-C" + clauseSeq;
            BudgetClause clause = new BudgetClause(null, clauseCode, clauseImport.getClauseNo(),
                    clauseImport.getTitle(), clauseImport.getClauseType(), clauseImport.getAmount(),
                    clauseImport.getTermCode());
            int deliverableSeq = 0;
            for (ClauseImport.DeliverableImport deliverableImport : clauseImport.getDeliverables()) {
                deliverableSeq++;
                BudgetDeliverable deliverable = new BudgetDeliverable(null,
                        clauseCode + "D" + deliverableSeq, deliverableImport.getTitle(), deliverableImport.getAmount());
                // 报价工作包回显为任务包（岗位/工时同源；编号仍由仓储生成 TP+雪花，指派人留空待 PM 补）
                for (ClauseImport.TaskImport taskImport : deliverableImport.getTasks()) {
                    BigDecimal hours = taskImport.getHours() == null || taskImport.getHours().compareTo(BigDecimal.ZERO) < 0
                            ? BigDecimal.ZERO : taskImport.getHours();
                    deliverable.tasksInternal().add(new TaskPackage(null, null,
                            taskImport.getName(), taskImport.getRoleCode(), null, hours));
                }
                clause.deliverablesInternal().add(deliverable);
            }
            this.clauses.add(clause);
        }
    }

    // ===== 交付包编制（refs #767：计划模板下拉选，导入之外的第二条创建路径） =====

    /**
     * 条款下新增交付包（编制口）：编制闸 + 条款存在 + 名称非空（取值域由前端计划模板下拉约束，
     * 与报价成本树同口径不做服务端模板校验）；code={条款编号}D{序号} 追加派生
     * （交付包无删除路径，size+1 不撞号）；金额可空。
     */
    public BudgetDeliverable addDeliverable(Long clauseId, String title, BigDecimal amount) {
        return this.addDeliverable(clauseId, title, amount, null);
    }
