/**
 * 费用分摊 HTTP payload 与应用命令、读取结果之间的适配器。
 */
public class TravelExpenseAllocationHttpConverter {

    public ReplacePlatformExpenseAllocationDraftCommand toReplacePlatformDraftCommand(
            Long travelApplicationId, Long groupId, ExpenseAllocationGroupDraftReqVO reqVO, Long operatorId) {
        return this.businessValidation(() -> new ReplacePlatformExpenseAllocationDraftCommand(travelApplicationId, groupId,
                requiredRequest(reqVO).getExpectedGroupVersion(), operatorId, this.toPlatformLines(reqVO.getLines())));
    }

    public SaveTravelExpenseAllocationCommand toSaveCommand(
            Long travelApplicationId, SaveTravelExpenseAllocationReqVO reqVO, Long operatorId) {
        SaveTravelExpenseAllocationReqVO request = this.requiredSaveRequest(reqVO);
        return this.businessValidation(() -> new SaveTravelExpenseAllocationCommand(travelApplicationId, operatorId,
                this.toPlatformDraftCommands(travelApplicationId, request.getPlatformDrafts(), operatorId),
                this.toManualDraftCommands(travelApplicationId, request.getManualDrafts(), operatorId),
                this.toAllowanceDraftCommands(travelApplicationId, request.getAllowanceDrafts(), operatorId)));
    }

    public ReplaceManualExpenseAllocationDraftCommand toReplaceManualDraftCommand(
            Long travelApplicationId, Long groupId, ExpenseAllocationGroupDraftReqVO reqVO, Long operatorId) {
        return this.businessValidation(() -> new ReplaceManualExpenseAllocationDraftCommand(travelApplicationId, groupId,
                requiredRequest(reqVO).getExpectedGroupVersion(), operatorId, this.toManualLines(reqVO.getLines())));
    }

    public ConfirmManualExpenseSelfPaidCommand toConfirmManualEmployeeSelfPaidCommand(
            Long travelApplicationId, Long groupId, Long expectedGroupVersion, Long operatorId) {
        return new ConfirmManualExpenseSelfPaidCommand(travelApplicationId, groupId, expectedGroupVersion, operatorId);
    }

    public ReplaceAllowanceExpenseAllocationDraftCommand toReplaceAllowanceDraftCommand(
            Long travelApplicationId, Long groupId, ExpenseAllocationGroupDraftReqVO reqVO, Long operatorId) {
        return this.businessValidation(() -> new ReplaceAllowanceExpenseAllocationDraftCommand(travelApplicationId, groupId,
                requiredRequest(reqVO).getExpectedGroupVersion(), operatorId, this.toAllowanceLines(reqVO.getLines())));
    }

    public ConfirmTravelExpenseAllocationCommand toConfirmCommand(
            Long travelApplicationId, ConfirmTravelExpenseAllocationReqVO reqVO, Long operatorId) {
        ConfirmTravelExpenseAllocationReqVO request = requiredConfirmRequest(reqVO);
        return new ConfirmTravelExpenseAllocationCommand(travelApplicationId, operatorId,
                request.getExpectedExpenseFactSnapshotHash(), request.getExpectedAllocationDraftSnapshotHash(),
                request.getTravelType(), request.getExpectedApplicationVersion());
    }

    public TravelExpenseAllocationRespVO toRespVO(TravelExpenseAllocationResult result) {
        if (result == null) {
            return null;

        respVO.setEditable(result.isEditable());
        respVO.setPostTripAllocationUnavailableReason(result.getPostTripAllocationUnavailableReason());
        respVO.setCurrentRevisionNo(result.getCurrentRevisionNo());
        respVO.setExpenseFactSnapshotHash(result.getExpenseFactSnapshotHash());
        respVO.setAllocationDraftSnapshotHash(result.getAllocationDraftSnapshotHash());
        respVO.setCollectionGroups(this.toGroupRespVOs(result.getCollectionGroups()));
        respVO.setBusinessObjectSummaries(this.toBusinessObjectRespVOs(result.getBusinessObjectSummaries()));
        respVO.setAllocationCandidates(this.toCandidateRespVOs(result.getAllocationCandidates()));
        respVO.setRejectedAllocationRows(result.getRejectedAllocationRows());

        List<cn.iocoder.yudao.module.fst.controller.admin.expense.travel.allocation.vo.ExpenseAllocationLogRespVO> excluded = new ArrayList<>();
        for (cn.iocoder.yudao.module.fst.domain.expense.travel.expenseallocation.ExpenseAllocationLog log : result.getExcludedSources()) {

    private List<PlatformExpenseAllocationLine> toPlatformLines(List<ExpenseAllocationLineReqVO> reqVOs) {
        List<PlatformExpenseAllocationLine> result = new ArrayList<>();
        for (ExpenseAllocationLineReqVO reqVO : requiredLines(reqVOs)) {
            result.add(PlatformExpenseAllocationLine.draft(reqVO.getBusinessObjectId(),
                    this.toBusinessObjectType(reqVO.getBusinessObjectType()), reqVO.getCostCenterId(),
                    this.toSecondaryAllocation(reqVO.getSecondaryAllocation()), reqVO.getAllocatedAmount(), reqVO.getNote()));
        }
        return result;
    }

    private List<ManualExpenseAllocationLine> toManualLines(List<ExpenseAllocationLineReqVO> reqVOs) {
        List<ManualExpenseAllocationLine> result = new ArrayList<>();
        for (ExpenseAllocationLineReqVO reqVO : requiredLines(reqVOs)) {
            result.add(ManualExpenseAllocationLine.draft(reqVO.getBusinessObjectId(),
                    this.toBusinessObjectType(reqVO.getBusinessObjectType()), reqVO.getCostCenterId(),
                    this.toSecondaryAllocation(reqVO.getSecondaryAllocation()), reqVO.getAllocatedAmount(), reqVO.getNote()));
        }
        return result;
    }

    private List<AllowanceExpenseAllocationLine> toAllowanceLines(List<ExpenseAllocationLineReqVO> reqVOs) {
        List<AllowanceExpenseAllocationLine> result = new ArrayList<>();
        for (ExpenseAllocationLineReqVO reqVO : requiredLines(reqVOs)) {
            result.add(AllowanceExpenseAllocationLine.draft(reqVO.getBusinessObjectId(),
                    this.toBusinessObjectType(reqVO.getBusinessObjectType()), reqVO.getCostCenterId(),
                    this.toSecondaryAllocation(reqVO.getSecondaryAllocation()), reqVO.getAllocatedAmount(), reqVO.getNote()));
        }
        return result;
    }

    private SecondaryAllocation toSecondaryAllocation(ExpenseAllocationSecondaryReqVO reqVO) {
        if (reqVO == null) {
            return null;
        }
        SecondaryAllocationType type = this.toSecondaryAllocationType(reqVO.getType());
        if (type == SecondaryAllocationType.CONTRACT_TERMS) {
            return SecondaryAllocation.contractTerm(reqVO.getReferencedId());
        }
        if (type == SecondaryAllocationType.OPPORTUNITY_PHASE) {
            return SecondaryAllocation.opportunityPhase(reqVO.getReferencedId());
        }
        throw new IllegalArgumentException("secondaryAllocation.type无效");
    }

    private BusinessObjectType toBusinessObjectType(String value) {
        for (BusinessObjectType type : BusinessObjectType.values()) {
            if (type.getValue().equals(value)) {
                return type;
            }
        }
        throw new IllegalArgumentException("businessObjectType无效");
    }

    private SecondaryAllocationType toSecondaryAllocationType(String value) {
        for (SecondaryAllocationType type : SecondaryAllocationType.values()) {
            if (type.getValue().equals(value)) {
                return type;
            }
        }
        throw new IllegalArgumentException("secondaryAllocation.type无效");
    }

    private List<ExpenseAllocationLineReqVO> requiredLines(List<ExpenseAllocationLineReqVO> lines) {
        if (lines == null || lines.contains(null)) {
            throw new IllegalArgumentException("lines不能为空且不能包含空项");
        }
        return lines;
    }
