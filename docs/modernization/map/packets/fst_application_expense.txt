/**
 * 组织全量同步应用编排：先准备启用部门，再协调员工，最后子到父退役部门。
 */
@Slf4j
@Service
public class TravelPlatformOrgSynchronizationUseCase {

    private final TravelPlatformOrgSynchronizationPreRunService preRunService;
    private final TravelPlatformOrgSynchronizationSource source;
    private final TravelPlatformOrgSynchronizationRemotePort remotePort;
    private final TravelPlatformOrgSynchronizationRepository repository;
    private final TravelPlatformEmployeeConsumptionTemplateConfigurationRepository templateConfigurationRepository;
    private final TravelPlatformEmployeeConsumptionTemplateRegistry templateRegistry;
    private final TravelPlatformEmployeeConsumptionTemplatePolicy templatePolicy;
    private final Clock clock;

    public TravelPlatformOrgSynchronizationUseCase(
            TravelPlatformOrgSynchronizationPreRunService preRunService,
            TravelPlatformOrgSynchronizationSource source,
            TravelPlatformOrgSynchronizationRemotePort remotePort,
            TravelPlatformOrgSynchronizationRepository repository,
            TravelPlatformEmployeeConsumptionTemplateConfigurationRepository templateConfigurationRepository,
            TravelPlatformEmployeeConsumptionTemplateRegistry templateRegistry,
            Clock clock) {
        if (preRunService == null || source == null || remotePort == null || repository == null
                || templateConfigurationRepository == null || templateRegistry == null || clock == null) {
            throw new IllegalArgumentException("组织同步编排依赖不能为空");
        }
        this.preRunService = preRunService;
        this.source = source;
        this.remotePort = remotePort;
        this.repository = repository;
        this.templateConfigurationRepository = templateConfigurationRepository;
        this.templateRegistry = templateRegistry;
        this.templatePolicy = new TravelPlatformEmployeeConsumptionTemplatePolicy();
        this.clock = clock;
    }

    /**
     * 在同一个运行证据与 durable fence 下执行部门和员工同步。
     * 未获得当前运行 owner 时返回空，避免调用方把历史运行误认作本次结果。
     */
    public Optional<TravelPlatformOrgSynchronizationRun> synchronize(
            String targetCompanyKey,
            Long sourceDepartmentId,
            TravelPlatformOrgSynchronizationTrigger trigger,
            String requestId,
            String traceId) {
        TravelPlatformOrgSynchronizationPreRunResult preparation = preRunService.tryPrepareRun(
                targetCompanyKey, trigger, requestId, traceId);
        if (!preparation.isOwner()) {
            // A non-owner did not execute this request and must not receive historical evidence as its result.
            return Optional.empty();
        }

        TravelPlatformOrgSynchronizationRun run = preparation.getOwnedRun().get();
        try {
            TravelPlatformOrgSynchronizationSnapshot snapshot;
            try {
                snapshot = source.readSnapshot(sourceDepartmentId);
                if (snapshot == null) {
                    throw new IllegalStateException("organization synchronization source snapshot is missing");
                }
            } catch (IllegalArgumentException | IllegalStateException ex) {
                return this.completeSourceFailure(run, TravelPlatformOrgSynchronizationStatus.REJECTED,
                        TravelPlatformOrgSynchronizationFailureCategory.INVALID_SOURCE, false, requestId, traceId);
            } catch (RuntimeException ex) {
                return this.completeSourceFailure(run, TravelPlatformOrgSynchronizationStatus.FAILED,
                        TravelPlatformOrgSynchronizationFailureCategory.SOURCE_UNAVAILABLE, true, requestId, traceId);
            }
            List<TravelPlatformOrgSynchronizationCandidate> departments = snapshot.getDepartmentCandidates();
            List<TravelPlatformOrgSynchronizationSourceMapping> sourceMappings =
                    repository.findSourceMappings(run.getTargetCompanyKey());
            Set<String> sourceMappingIdentities = this.sourceMappingIdentities(sourceMappings);
            List<TravelPlatformOrgSynchronizationCandidate> enabledDepartments =
                    this.activeOrInvalidCandidates(departments);
            List<TravelPlatformOrgSynchronizationCandidate> retiringDepartments =
                    this.mappedInactiveCandidates(departments, sourceMappingIdentities);
            List<TravelPlatformOrgSynchronizationCandidate> sourceEmployees = snapshot.getEmployeeCandidates();
            // 只筛选本轮会进入员工阶段的活动或已纳管停用员工，避免无关历史停用行依赖模板目录。
            List<TravelPlatformOrgSynchronizationCandidate> employees =
                    this.activeAndMappedInactiveCandidates(sourceEmployees, sourceMappingIdentities);
            // 仅可下发员工需要消费模板；来源异常员工保留既有异常证据链。
            if (this.hasDispatchableCandidate(employees)) {
                List<TravelPlatformEmployeeConsumptionTemplate> catalogue;
                // 模板目录也是 provider I/O，必须与读写员工共用同一执行权围栏。
                preRunService.assertProviderIoAllowed(run.getTargetCompanyKey(), run);
                try {
                    catalogue = templateRegistry.listTemplates();
                } catch (IllegalArgumentException ex) {
                    return this.completeSourceFailure(run, TravelPlatformOrgSynchronizationStatus.REJECTED,
                            TravelPlatformOrgSynchronizationFailureCategory.INVALID_SOURCE, false, requestId, traceId);
                } catch (IllegalStateException ex) {
                    return this.completeSourceFailure(run, TravelPlatformOrgSynchronizationStatus.FAILED,
                            TravelPlatformOrgSynchronizationFailureCategory.SOURCE_UNAVAILABLE, true, requestId, traceId);
                } catch (RuntimeException ex) {
                    return this.completeSourceFailure(run, TravelPlatformOrgSynchronizationStatus.FAILED,
                            TravelPlatformOrgSynchronizationFailureCategory.SOURCE_UNAVAILABLE, true, requestId, traceId);
                }
                List<TravelPlatformEmployeeConsumptionTemplateConfiguration> configurations;
                List<TravelPlatformEmployeeConsumptionTemplateMapping> mappings;
                try {
                    configurations = templateConfigurationRepository.findActiveConfigurations();
                    mappings = templateConfigurationRepository.findActiveMappings();
                } catch (RuntimeException ex) {
                    return this.completeSourceFailure(run, TravelPlatformOrgSynchronizationStatus.FAILED,
                            TravelPlatformOrgSynchronizationFailureCategory.SOURCE_UNAVAILABLE, true, requestId, traceId);
                }
                try {
                    employees = templatePolicy.resolveEmployees(employees, configurations, mappings, catalogue);
                } catch (IllegalArgumentException | IllegalStateException ex) {
                    return this.completeSourceFailure(run, TravelPlatformOrgSynchronizationStatus.REJECTED,
                            TravelPlatformOrgSynchronizationFailureCategory.INVALID_SOURCE, false, requestId, traceId);
                }
            }
            // 输出本次来源和纳管范围摘要，便于从运行日志核验首次与后续对账的输入边界。
            this.logSourceSnapshot(run, snapshot, departments, sourceEmployees, sourceMappings,
                    enabledDepartments, retiringDepartments);
            List<TravelPlatformOrgSynchronizationFailureEvidence> warningEvidence =
                    this.recordSourceWarnings(run, snapshot);
            SynchronizationStage departmentStage = this.synchronizeEnabledDepartments(
                    run, enabledDepartments, departments, trigger);
            Set<String> verifiedEmployeeDepartmentThirdIds = new HashSet<>(departmentStage.getVerifiedThirdIds());
            // The PSA source anchor is the configured counterpart of the provider-owned root, not a candidate.
            verifiedEmployeeDepartmentThirdIds.add(String.valueOf(sourceDepartmentId));
            SynchronizationStage employeeStage = this.synchronizeEmployees(
                    run, employees,
                    verifiedEmployeeDepartmentThirdIds, trigger);
            SynchronizationStage retirementStage = this.synchronizeRetiringDepartments(
                    run, departments, retiringDepartments, snapshot);
            return this.complete(run, departmentStage, employeeStage, retirementStage,
                    warningEvidence, requestId, traceId);
        } finally {
            preRunService.release(targetCompanyKey);
        }
    }
