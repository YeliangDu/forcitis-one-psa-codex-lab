/**
 * 期初数据体检实现。设计要点：
 * <ul>
 *   <li><b>只读</b>：扫两个抽取视图取行，检测委托过账前置服务，不写库、不动期间进度；</li>
 *   <li><b>工时口径</b>：{@link ManHourPostingCheckService}（审批 / 费率九件套 / type·wpt 联动 / 成本归属）；</li>
 *   <li><b>费用口径</b>：{@link ExpensePostingCheckService#checkRows}（请求内预热配置/档案后内存九条判定）；</li>
 *   <li><b>商机状态</b>：工时参考量按档案 {@code won_date} vs 发生日还原当时关闭状态；
 *       费用不改写视图快照，由过账检测把当时状态与 {@code psa_cost_summary_new.close_status} 比对；</li>
 *   <li><b>期间窗口</b>：抽数一律按自然月（1 号～月末），不取期间表起止日；期间表字段仍回填给执行闸，不进入体检结论；</li>
 *   <li><b>下钻</b>：同月回查视图 → 再跑同一过账检测 → 按问题类型过滤分页；</li>
 *   <li><b>成块抽数</b>：{@link #check} 按每块 {@value #MONTHS_PER_SCAN} 个月抽一轮数，块内共享商机档案与
 *       工时/费用过账检测，再按行日期分桶回各月；块间并发上限 {@value #MAX_PARALLELISM}。</li>
 * </ul>
 */
@Service
public class OpeningHealthCheckUseCase implements IOpeningHealthCheckUseCase {

    /**
     * 单次抽数覆盖的月数。两个抽取视图每次扫描都有一笔与行数无关的固定开销（dev 实测约 1.1s/次），
     * 逐月抽数时这笔开销按月数线性叠加；成块后按块摊薄。块内驻留行数随月数上升，与并发上限配套定 6。
     */
    private static final int MONTHS_PER_SCAN = 6;
    /** 块间并发上限：各块抢的是库侧同一份视图扫描，再高只增争抢；块变大后也要压住内存峰值。 */
    private static final int MAX_PARALLELISM = 3;
    /** 工时 checkBatch 分片，避免一次 IN 过大。 */
    private static final int TS_CHECK_BATCH = 500;
    /** 商机档案预热 IN 分片：SQL Server 单语句 2100 参数上限，块内 distinct 商机可达数千。 */
    private static final int OPP_IN_BATCH = 1000;

    private final CostSummaryTimesheetViewMapper timesheetMapper;
    private final CostSummaryViewMapper summaryMapper;
    private final AccountingPeriodService accountingPeriodService;
    private final ManHourPostingCheckService manHourPostingCheckService;
    private final ExpenseHealthCheckResolver expenseHealthCheckResolver;
    private final OpportunityMapper opportunityMapper;
    private final ProjectTermArchiveLoader projectTermArchiveLoader;

    public OpeningHealthCheckUseCase(CostSummaryTimesheetViewMapper timesheetMapper,
                                     CostSummaryViewMapper summaryMapper,
                                     AccountingPeriodService accountingPeriodService,
                                     ManHourPostingCheckService manHourPostingCheckService,
                                     ExpensePostingCheckService expensePostingCheckService,
                                     OpportunityMapper opportunityMapper,
                                     ProjectTermArchiveLoader projectTermArchiveLoader) {
        this.timesheetMapper = timesheetMapper;
        this.summaryMapper = summaryMapper;
        this.accountingPeriodService = accountingPeriodService;
        this.manHourPostingCheckService = manHourPostingCheckService;
        this.expenseHealthCheckResolver = new ExpenseHealthCheckResolver(expensePostingCheckService);
        this.opportunityMapper = opportunityMapper;
        this.projectTermArchiveLoader = projectTermArchiveLoader;
    }

    @Override
    public List<OpeningMonthHealthResult> check(String companyCode, YearMonth start, YearMonth end) {
        if (companyCode == null || companyCode.trim().isEmpty() || start == null || end == null) {
            throw new ServiceException(400, "参数缺失：companyCode/startPeriod/endPeriod 均必填");
        }
        // 体检只读探数：不卡 OPENING_MIN/MAX，仅要求起止有序
        if (end.isBefore(start)) {
            throw new ServiceException(400, "期初体检期间起止须有序（endPeriod ≥ startPeriod）");
        }
        List<YearMonth> months = this.listMonths(start, end);
        List<List<YearMonth>> blocks = partitionMonths(months, MONTHS_PER_SCAN);
        // 项目条款档案整次体检取一次（2.3 万行 / 一条 SQL），并发分块与工时费用两侧共用同一份不可变上下文
        ProjectTermArchive archive = projectTermArchiveLoader.loadAll();
        return blocks.size() == 1
                ? this.checkBlock(companyCode, blocks.get(0), archive)
                : this.checkBlocksParallel(companyCode, blocks, archive);
    }

    private List<OpeningMonthHealthResult> checkBlocksParallel(String companyCode, List<List<YearMonth>> blocks,
                                                               ProjectTermArchive archive) {
        int poolSize = Math.min(blocks.size(), MAX_PARALLELISM);
        ExecutorService pool = Executors.newFixedThreadPool(poolSize);
        try {
            List<Future<List<OpeningMonthHealthResult>>> futures = new ArrayList<>(blocks.size());
            for (List<YearMonth> block : blocks) {
                futures.add(pool.submit(() -> this.checkBlock(companyCode, block, archive)));
            }
            List<OpeningMonthHealthResult> out = new ArrayList<>();
            for (Future<List<OpeningMonthHealthResult>> f : futures) {
                try {
                    out.addAll(f.get());
                } catch (InterruptedException ie) {
                    Thread.currentThread().interrupt();
                    throw new ServiceException(500, "期初体检被中断");
                } catch (ExecutionException ee) {
                    Throwable cause = ee.getCause() != null ? ee.getCause() : ee;
                    if (cause instanceof ServiceException) {
                        throw (ServiceException) cause;
                    }
                    throw new ServiceException(500, "期初体检并行分块失败：" + cause.getMessage());
                }
            }
            return out;
        } finally {
            pool.shutdownNow();
        }
    }

    private static List<YearMonth> listMonths(YearMonth start, YearMonth end) {
        List<YearMonth> months = new ArrayList<>();
        for (YearMonth ym = start; !ym.isAfter(end); ym = ym.plusMonths(1)) {
            months.add(ym);
        }
        return months;
    }

    private static List<List<YearMonth>> partitionMonths(List<YearMonth> months, int size) {
        List<List<YearMonth>> chunks = new ArrayList<>();
        for (int i = 0; i < months.size(); i += size) {
            chunks.add(new ArrayList<>(months.subList(i, Math.min(i + size, months.size()))));
        }
        return chunks;
    }
