/**
 * 财务修正清单导入（refs #1176）。
 *
 * <p>上传即建批次并返回批次号，18 万行的解析放后台跑；页面到「导入批次」页手动刷新看进度。
 *
 * <p>清单里只读两个 sheet：
 * <ul>
 *   <li><b>全部</b> —— 有明细ID 且与 OA 原值有差异 → {@code excel_update}；
 *       无明细ID → {@code excel_insert}（财务补录、未走系统的报销）；
 *       有明细ID 但与原值完全一致 → 不落库（实测 13.2 万行属此类）</li>
 *   <li><b>删除</b> —— 按明细ID 标记 {@code excel_del}，整行快照从 OA 原值搬</li>
 * </ul>
 * 其余 sheet 一概忽略。
 *
 * <p><b>为什么自持线程池而不用 {@code @Async} + 自定义 Executor Bean</b>：
 * Spring Boot 的 {@code TaskExecutionAutoConfiguration#applicationTaskExecutor} 带
 * {@code @ConditionalOnMissingBean(Executor.class)}——容器里一旦出现任何 Executor Bean，
 * 该自动配置就失效，而 Flowable 硬依赖它，会导致整个应用起不来。
 */
@Service
@Slf4j
public class CostCorrectionImportService {

    /** 只关注这两个 sheet。 */
    private static final String SHEET_MAIN = "全部";
    private static final String SHEET_DELETE = "删除";
    /** 每块行数：小了事务次数多，大了单事务过重、进度刷新迟钝。 */
    private static final int CHUNK_SIZE = 2000;
    /** 错误行上限：错误行全量驻留内存，防止一个格式全错的文件把堆吃光。 */
    private static final int MAX_ERROR_ROWS = 100_000;
    /** 并发度压到 2：导入是重 IO + 大内存（基座快照 18 万行常驻），过高会互相挤兑。 */
    private static final int MAX_THREADS = 2;
    private static final int QUEUE_CAPACITY = 16;
    /** 基座视图里核算组织恒为该值，补录行沿用。 */
    private static final String DEFAULT_COMPANY_CODE = "1001";

    private static final DateTimeFormatter BATCH_TIME = DateTimeFormatter.ofPattern("yyyyMMddHHmmss");
    private static final AtomicInteger SEQ = new AtomicInteger();

    private ThreadPoolExecutor executor;

    @Resource

    @Resource

    @Resource

    @PostConstruct
    void initExecutor() {
        AtomicInteger seq = new AtomicInteger();
        executor = new ThreadPoolExecutor(1, MAX_THREADS, 120L, TimeUnit.SECONDS,
                new ArrayBlockingQueue<>(QUEUE_CAPACITY),
                r -> {
                    Thread t = new Thread(r, "cost-correction-import-" + seq.incrementAndGet());
                    t.setDaemon(true);
                    return t;
                },
                new ThreadPoolExecutor.AbortPolicy());
    }

    @PreDestroy
    void shutdownExecutor() {
        if (executor == null) {
            return;
        }
        executor.shutdown();
        try {
            if (!executor.awaitTermination(60, TimeUnit.SECONDS)) {
                executor.shutdownNow();
            }
        } catch (InterruptedException e) {
            executor.shutdownNow();
            Thread.currentThread().interrupt();
        }
    }

    /**
     * 提交一次导入；本方法毫秒级返回，不阻塞 HTTP 线程。
     *
     * @return 批次号，供页面查进度
     */
    public String submit(String fileName, byte[] content, String operator) {
        String batchNo = "CCI" + LocalDateTime.now().format(BATCH_TIME)
                + String.format("%03d", SEQ.updateAndGet(v -> (v + 1) % 1000));
        Long batchId = createBatch(batchNo, fileName, content, operator);
        log.info("[submit][批次({}) 已建立，文件={}，大小={}B]", batchNo, fileName, content.length);
        try {
            executor.execute(() -> {
                try {
                    doRun(batchId, batchNo, fileName, content, operator);
                } catch (Exception e) {
                    log.error("[submit][批次({}) 导入失败]", batchNo, e);
                    markFailed(batchId, CostCorrectionFailure.describe("导入处理", e));
                }
            });
        } catch (RejectedExecutionException e) {
            // 队列已满：明确置失败，好过静默丢弃一次已建批次的导入
            log.warn("[submit][批次({}) 被线程池拒绝，队列已满]", batchNo);
            markFailed(batchId, "导入任务队列已满，请等待在途批次完成后重试");
        }
        return batchNo;
    }

    /**
     * 成本中心中文名 → 部门ID 候选集。
     *
     * <p><b>只认「过账门禁认可的那一套部门」</b>——即 {@code deptService.getDeptInfoMap()}，
     * 与 {@code ExpensePostingCheckServiceImpl#checkCostCenter} 校验用的是同一个集合

     * 换任何别的来源都会出现「导入成功但过账依然不通过」：写进去一个门禁不认的部门ID，
     * 履历上显示改了，门禁照样拦，等于白改还污染了数据。
     *
     * <p><b>返回候选列表而非单值</b>：部门名大量重名（195 个名称对应 367 条记录，
     * 实测补录清单里 2637 行撞在 19 个重名上）。用 Map&lt;name, id&gt; 会让后写的静默覆盖前面的，
     * 等于随机挑一个部门写进财务数据——必须让调用方看见「有几个候选」并按歧义处理。
     */
    private Map<String, List<Long>> loadDeptByName() {
        Map<String, List<Long>> map = new HashMap<>();
        for (Map.Entry<Long, DeptRespDTO> e : deptService.getDeptInfoMap().entrySet()) {
            DeptRespDTO d = e.getValue();
            if (d == null || d.getName() == null) {
                continue;
            }
            List<Long> ids = map.computeIfAbsent(d.getName().trim(), k -> new ArrayList<>());
            if (!ids.contains(e.getKey())) {
                ids.add(e.getKey());
            }
        }
        return map;
    }
