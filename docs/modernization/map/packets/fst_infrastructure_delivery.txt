/**
 * 项目预算仓储实现：项目主单复用存量 ProjectMapper（psa_pm_project 唯一真相源），
 * WBS 三表（条款/交付包/任务包）经新建 Mapper 批量装载组树；DO ↔ 聚合转换收在本类。
 * <p>锁定标记读写 wbs_budget_locked（#619 增列），存量 budget_status 归存量预算体系、零触碰。
 */
@Slf4j
@Component
public class ProjectBudgetRepositoryImpl implements ProjectBudgetRepository {

    /**
     * 当前基线版本号（refs #1106）：本期基线单版本，预算变更就地改写当前版本，
     * 历史留档由 psa_project_budget_snapshot 承担；多版本并存待基线版本对照需求出现再启用。
     */
    private static final int CURRENT_REVISION_NO = 1;

    @Resource
    private ProjectMapper projectMapper;
    @Resource
    private PositionRateQuery positionRateQuery;
    @Resource
    private QuotationMapper quotationMapper;
    @Resource
    private ContractClauseMapper contractClauseMapper;
    @Resource
    private DeliverableMapper deliverableMapper;
    @Resource
    private TaskPackageMapper taskPackageMapper;
    @Resource
    private BudgetDeliverableMapper budgetDeliverableMapper;
    @Resource
    private BudgetTaskPackageMapper budgetTaskPackageMapper;
    @Resource
    private BudgetCostItemMapper budgetCostItemMapper;
    @Resource
    private TimeSheetRateService timeSheetRateService;

    @Override
    public ProjectBudget findByProjectId(Long projectId) {
        ProjectDO projectDO = projectId == null ? null : projectMapper.selectById(projectId);
        if (projectDO == null) {
            return null;
        }
        // 条款层共用（A8 档案投影），交付包/任务包读**基线表**（refs #1106）——
        // 执行期在进度页补拆的节点只在执行表，不会污染这棵已审批的基线树
        List<ContractClauseDO> clauseRows = contractClauseMapper.selectListByProjectId(projectId);
        List<Long> clauseIds = clauseRows.stream().map(ContractClauseDO::getId).collect(Collectors.toList());
        List<BudgetDeliverableDO> deliverableRows =
                budgetDeliverableMapper.selectListByClauseIds(clauseIds, CURRENT_REVISION_NO);
        List<Long> deliverableIds = deliverableRows.stream()
                .map(BudgetDeliverableDO::getId).collect(Collectors.toList());
        List<BudgetTaskPackageDO> taskRows = budgetTaskPackageMapper.selectListByDeliverableIds(deliverableIds);

        Map<Long, List<BudgetDeliverableDO>> deliverablesByClause = deliverableRows.stream()
                .collect(Collectors.groupingBy(BudgetDeliverableDO::getClauseId));
        Map<Long, List<BudgetTaskPackageDO>> tasksByDeliverable = taskRows.stream()
                .collect(Collectors.groupingBy(BudgetTaskPackageDO::getDeliverableId));

        List<BudgetClause> clauses = new ArrayList<>();
        for (ContractClauseDO clauseDO : clauseRows) {
            // termCode 与 obsolete 必须装载（refs #976/#991）：前者是变更时与档案条款比对的锚，
            // 缺失会让整棵树被判「档案中已消失」；后者缺失会让失效节点在聚合里恢复成可写
            BudgetClause clause = new BudgetClause(clauseDO.getId(), clauseDO.getCode(), clauseDO.getClauseNo(),
                    clauseDO.getTitle(), clauseDO.getClauseType(), clauseDO.getAmount(),
                    clauseDO.getTermCode(), Boolean.TRUE.equals(clauseDO.getObsolete()));
            for (BudgetDeliverableDO deliverableDO : deliverablesByClause
                    .getOrDefault(clauseDO.getId(), Collections.emptyList())) {
                BudgetDeliverable deliverable = new BudgetDeliverable(deliverableDO.getId(), deliverableDO.getCode(),
                        deliverableDO.getTitle(), deliverableDO.getAmount(),
                        Boolean.TRUE.equals(deliverableDO.getObsolete()));
                // 来源选项 id（refs #1061）：title 只是名称快照，本值才是工作包下拉与引用统计的关联锚
                deliverable.restoreSourceDeliverableId(deliverableDO.getSourceDeliverableId());
                for (BudgetTaskPackageDO taskDO : tasksByDeliverable
                        .getOrDefault(deliverableDO.getId(), Collections.emptyList())) {
                    // 基线无指派（编制只到岗位，refs #1083）：派工是执行表的事
                    TaskPackage task = TaskPackage.restore(taskDO.getId(), taskDO.getCode(), taskDO.getName(),
                            taskDO.getRoleCode(), null, taskDO.getPlannedHours());
                    task.restoreObsolete(Boolean.TRUE.equals(taskDO.getObsolete()));
                    task.restoreSourceWorkpkgId(taskDO.getSourceWorkpkgId());
                    // 冻结费率只在锁定态回装（refs #1824）：解锁编辑期按实时费率显示（重锁时整体覆盖），
                    // 不清列——变更被驳回 / 撤回仍停在编辑态，列值等下一次锁定覆盖即可
                    if (Boolean.TRUE.equals(projectDO.getWbsBudgetLocked())) {
                        task.restoreLockedCostRate(taskDO.getLockedCostRate());
                    }
                    deliverable.restoreTask(task);
                }
                clause.restoreDeliverable(deliverable);
            }
            clauses.add(clause);
        }
        ProjectBudget budget = ProjectBudget.restore(projectDO.getId(), projectDO.getCode(),
                projectDO.getOpportunityId(), projectDO.getAmount(),
                Boolean.TRUE.equals(projectDO.getWbsBudgetLocked()),
                projectDO.getWbsBudgetApprovalStatus(), projectDO.getWbsBudgetProcessId(), clauses);
        // 费用类目装载（refs #748；LinkedHashMap 保插入序）
        java.util.Map<String, java.math.BigDecimal> items = new java.util.LinkedHashMap<>();
        for (BudgetCostItemDO itemDO : budgetCostItemMapper.selectListByProjectId(projectId)) {
            items.put(itemDO.getCategory(), itemDO.getAmount());
        }
        budget.restoreCostItems(items);
        // 费率取数城市（refs #1826）：项目「预算所属城市」优先，空时回退预算来源报价城市，再空为不过滤
        budget.restoreCityCategoryCode(this.resolveBudgetCityCategory(projectDO));
        this.applyRoleCostRates(budget);
        return budget;
    }

    /**
     * 装载后注入岗位标准成本费率（refs #1083，与报价成本树同源：费率主数据 salary_grade_no='0' 岗位级口径）：
     * 全量岗位费率一次查出按 roleCode 命中；未命中留 null（成本计 0，前端费率列提示）。
     * <p>城市维度（refs #1146/#1826）：按聚合装载时解析的城市取行（项目列优先 → 报价 → 不过滤），
     * 保证预算与报价同岗同价。
     */
    private void applyRoleCostRates(ProjectBudget budget) {
        java.util.Map<String, java.math.BigDecimal> rateByRole =
                this.resolveRoleCostRatesByCity(budget.getCityCategoryCode());
        for (BudgetClause clause : budget.getClauses()) {
            for (BudgetDeliverable deliverable : clause.getDeliverables()) {
                for (TaskPackage task : deliverable.getTasks()) {
                    task.applyRoleCostRate(rateByRole.get(task.getRoleCode()));
                }
            }
        }
    }

    /**
     * 预算费率取数城市（refs #1826）：项目主表「预算所属城市」优先；为空回退预算来源报价的城市。
     * 收在装载点解析：费率注入被详情/锁定/任务包 CRUD 多路径共用，才能保证各路径同价。
     */
    private String resolveBudgetCityCategory(ProjectDO projectDO) {
        String projectCity = projectDO.getCityCategoryCode();
        if (projectCity != null && !projectCity.trim().isEmpty()) {
            return projectCity.trim();
        }
        return this.resolveBudgetCityCategory(projectDO.getOpportunityId());
    }

    /**
     * 预算来源报价的城市（refs #1146，选取口径见 {@link BudgetSourceQuotationPicker}）：
     * 无报价或报价未选城市返回 null（费率回退不过滤口径）。
     */
    private String resolveBudgetCityCategory(Long opportunityId) {
        if (opportunityId == null) {
            return null;
        }
        QuotationDO source = BudgetSourceQuotationPicker.pick(
                quotationMapper.selectListByOpportunityId(opportunityId));
        return source == null ? null : source.getCityCategoryCode();
    }
