    /** 敏感成本可见权限点（refs #748；评审 M1：服务端脱敏唯一权威，前端打码仅展示层） */
    private static final String COST_VIEW_PERMISSION = "fst:pm-project:budget-cost-view";

    @GetMapping("/get")
    @ApiOperation("项目预算详情（摘要+三级树+报价对照；PM_BUDGET 审批视图共用）")
    @RateLimiter(name = "backendA")
    // 可见性：有查询权 或 是本单审批参与人（refs #1378 S1）。旧口径整体不设闸，合同金额、条款
    // 拆分金额、人力成本合计与每个任务包的岗位标准成本费率对全体登录用户敞开、可按 projectId 遍历。
    // 预算按项目主键定位，行级过滤显式关闭
    @DataPermission(enable = false)
    public CommonResult<PmProjectBudgetDetailRespVO> getBudget(@RequestParam("projectId") Long projectId) {
        ProjectBudgetDetailResult result = getProjectBudgetUseCase.execute(projectId);
        approvalVisibilityGuard.requireView("fst:pm-project:query",
                result.getBudget() == null ? null : result.getBudget().getProcessInstanceId(),
                ErrorCodeConstants.PROJECT_NOT_EXISTS);
        return success(this.toDetailVO(result));
    }

    @PostMapping("/import")
    @ApiOperation("初始化项目预算（条款层取自项目档案条款，交付层底稿取自赢单报价按条款编号挂载；已初始化拒绝）")
    @RateLimiter(name = "backendA")
    @PreAuthorize("@ss.hasPermission('fst:pm-project:budget')")
    // 写路径按项目主键定位，锁定/幂等闸在聚合内，显式关闭行级过滤
    @DataPermission(enable = false)
    public CommonResult<PmProjectBudgetInitRespVO> initializeBudget(@RequestParam("projectId") Long projectId) {
        InitializeBudgetResult result = initializeProjectBudgetUseCase.execute(projectId);
        PmProjectBudgetInitRespVO respVO = new PmProjectBudgetInitRespVO();
        respVO.setClauseCount(result.getClauseCount());
        respVO.setDeliverableCount(result.getDeliverableCount());
        respVO.setTaskCount(result.getTaskCount());
        respVO.setSourceQuotationCode(result.getSourceQuotationCode());
        respVO.setUnmatchedClauses(result.getUnmatchedClauses().stream().map(unmatched -> {
            PmProjectBudgetInitRespVO.UnmatchedClauseVO unmatchedVO =
                    new PmProjectBudgetInitRespVO.UnmatchedClauseVO();
            unmatchedVO.setClauseCode(unmatched.getClauseCode());
            unmatchedVO.setTitle(unmatched.getTitle());
            unmatchedVO.setDeliverableCount(unmatched.getDeliverableCount());
            return unmatchedVO;
        }).collect(java.util.stream.Collectors.toList()));
        return success(respVO);
    }

    // 同上：写路径按主键定位，闸在聚合内

    public CommonResult<Long> createTaskPackage(@Valid @RequestBody PmProjectTaskPackageSaveReqVO reqVO) {
        return success(createTaskPackageUseCase.execute(new CreateTaskPackageCommand(reqVO.getProjectId(),
                reqVO.getDeliverableId(), reqVO.getName(), reqVO.getRoleCode(),
                reqVO.getPlannedHours())));
    }

    @PostMapping("/task/update")
    @ApiOperation("修改任务包（名称/岗位/指派/预算工时）")
    @RateLimiter(name = "backendA")
    @PreAuthorize("@ss.hasPermission('fst:pm-project:budget')")
    // 同上
    @DataPermission(enable = false)
    public CommonResult<Boolean> updateTaskPackage(@Valid @RequestBody PmProjectTaskPackageSaveReqVO reqVO) {
        updateTaskPackageUseCase.execute(new UpdateTaskPackageCommand(reqVO.getProjectId(), reqVO.getId(),
                reqVO.getName(), reqVO.getRoleCode(), reqVO.getPlannedHours(),
                reqVO.getSourceWorkpkgId()));
        return success(true);
    }

    @DeleteMapping("/task/delete")
    @ApiOperation("删除任务包（软删）")
    @RateLimiter(name = "backendA")

    @DataPermission(enable = false)
    public CommonResult<Boolean> deleteTaskPackage(@RequestParam("projectId") Long projectId,

    }

    @PostMapping("/lock")
    @ApiOperation("保存预算（提交 PM_BUDGET 内部审批，通过后锁定；审批中所有编制操作拒绝；"

                                            @RequestParam(value = "obsoleteConfirmed", required = false,
                                                    defaultValue = "false") Boolean obsoleteConfirmed) {
        lockProjectBudgetUseCase.execute(projectId, Boolean.TRUE.equals(obsoleteConfirmed));
        return success(true);
    }
