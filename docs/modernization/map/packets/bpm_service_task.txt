    @Override
    public PageResult<BpmTaskTodoPageItemRespVO> getTodoTaskPage(Long userId, BpmTaskTodoPageReqVO pageVO) {

        // 查询待办任务
        TaskQuery taskQuery = taskService.createTaskQuery().taskAssignee(String.valueOf(userId)) // 分配给自己
                .orderByTaskCreateTime().desc(); // 创建时间倒序
        this.buildTaskQuery(pageVO, taskQuery);

        // 行审批(逐行)一行一个任务，须按「审批合并」聚成一条待办（表单+节点+审批人 = 一屏、一次性审完）。
        // 收拢改变条目数 → 不能再用 Flowable 的 listPage/count 按 task 分页：总数会对不上，且同一节点的
        // 行任务会被页边界撕成两页、同一条待办出现两次。故全量拉本人待办 → 收拢 → 内存分页。
        // 量级：单人待办（数十条级）可接受；跨人的 getTodoTaskPage(pageReqVO) 不走此路径。
        List<Task> mergedTasks = this.mergeRowApprovalTasks(taskQuery.list());

        // #920 知会并集：待办 = 存活审批任务 ∪ 未读知会。知会任务被级联自动办结、不在 Flowable 运行时，
        //   故从 ext 表并入（result=知会 且 read_time 为空）；打开流程详情打点后自然消失。
        List<BpmTaskTodoPageItemRespVO> notifyItems = this.buildUnreadNotifyItems(userId, pageVO);

        int total = mergedTasks.size() + notifyItems.size();
        int fromIndex = PageUtils.getStart(pageVO);
        if (fromIndex >= total) {
            return PageResult.empty(total);
        }
        List<BpmTaskTodoPageItemRespVO> taskItems = mergedTasks.isEmpty()
                ? new ArrayList<>()
                : this.buildTask(mergedTasks, mergedTasks.size()).getList();
        taskItems.forEach(item -> item.setDataType("PROCESS_TODO"));
        this.fillH5ApprovalRoute(taskItems, mergedTasks);
        // 待办列表与任务详情共用模型拓扑事实，供发起人识别退回修改入口。
        this.fillTodoReworkEntry(taskItems, mergedTasks);
        List<BpmTaskTodoPageItemRespVO> merged = new ArrayList<>(taskItems);
        merged.addAll(notifyItems);
        merged.sort(Comparator.comparing(BpmTaskTodoPageItemRespVO::getCreateTime,
                Comparator.nullsLast(Comparator.reverseOrder())));
        int toIndex = Math.min(fromIndex + pageVO.getPageSize(), total);
        return new PageResult<>(new ArrayList<>(merged.subList(fromIndex, toIndex)), total);
    }

    /**
     * 行审批（逐行）：同一 (流程实例, 节点) 的多个行任务在待办里聚成<b>一条</b>，保留首个作代表。
     *
     * <p>审批人对这些行是「一屏、一次性审完」（后端 {@link #decide} 强制覆盖本节点全部待办行），
     * 故待办列表按一次审批动作计数，而非按 task 计数。非行审批任务原样保留，行为不变。
     *
     * <p>{@code isRowApprovalNode} 每次要查分配规则，此处按 (流程定义, 节点) memo，避免待办列表逐条打库
     * （本接口被前端定时轮询）。
     */
    @VisibleForTesting
    List<Task> mergeRowApprovalTasks(List<Task> tasks) {
        if (CollUtil.isEmpty(tasks)) {
            return Collections.emptyList();
        }
        List<Task> result = new ArrayList<>(tasks.size());
        Map<String, Boolean> rowApprovalMemo = new HashMap<>();
        Set<String> seenRowGroups = new HashSet<>();
        for (Task task : tasks) {
            Boolean rowApproval = rowApprovalMemo.computeIfAbsent(
                    task.getProcessDefinitionId() + ":" + task.getTaskDefinitionKey(),
                    key -> this.isRowApprovalNode(task));
            if (!Boolean.TRUE.equals(rowApproval)) {
                result.add(task);
                continue;
            }
            if (seenRowGroups.add(task.getProcessInstanceId() + ":" + task.getTaskDefinitionKey())) {
                result.add(task); // 本组代表：点进去后审批页按 (节点+审批人) 重新聚齐全部行
            }
        }
        return result;
    }
