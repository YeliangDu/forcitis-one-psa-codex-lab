    @Override
    public PageResult<ProjectPageItemRespVO> getUserPage(ProjectPageReqVO reqVO) {
        Long userId = SecurityFrameworkUtils.getLoginUserId();
        List<UserRoleDO> userRoleDOList = userRoleMapper.selectList(UserRoleDO::getUserId, userId);
        List<Long> userRoleIdList = CollectionUtils.convertListSingle(userRoleDOList, UserRoleDO::getRoleId);
        boolean modificationRights = userRoleIdList.contains(FSTConstant.Role.BU项目总监);

        PageResult<ProjectBO> pageResult = this.getProjectPage(reqVO);
        List<ProjectBO> projectDOList = pageResult.getList();
        PageResult<ProjectPageItemRespVO> result = this.buildProject(projectDOList, pageResult.getTotal());
        for (ProjectPageItemRespVO projectPageItemRespVO : result.getList()) {
            if (userId != null && (userId.equals(projectPageItemRespVO.getDeliveryManagerId())
                    || userId.equals(projectPageItemRespVO.getManagerId())
                    || modificationRights))
                projectPageItemRespVO.setModificationRights(true);
        }
        return result;
    }

    @Override
    public List<ProjectSimpleListRespVO> getSimpleList(ProjectListReqVO reqVO) {
        // 所有用户
        Map<Long, UserInfoRespVO> userMap = userService.getUserMapByCache();
        // 所有部门
        Map<Long, DeptRespDTO> deptMap = CollectionUtils.convertMap(deptApi.getDeptsByCache().getData(), DeptRespDTO::getId);
        LambdaQueryWrapperX<ProjectDO> queryWrapper = new LambdaQueryWrapperX<>();

        if (StrUtils.isNotEmpty(reqVO.getProjectKey())) {
            queryWrapper.likeIfPresent(ProjectDO::getCode, reqVO.getProjectKey());
            queryWrapper.or().like(ProjectDO::getName, reqVO.getProjectKey());
        }
        List<ProjectDO> projectDOList = projectMapper.getTop(queryWrapper, LambdaUtil.getField(ProjectDO::getInitiationDate), 50, ProjectDO.class);

        List<ProjectSimpleListRespVO> projectSimpleListRespVOList = ProjectConvert.INSTANCE.convert1(projectDOList);
        List<Long> customerIdList = CollectionUtils.convertListSingle(projectSimpleListRespVOList, ProjectSimpleListRespVO::getCustomerId);
        Map<Long, CustomerDO> customerMap = CollectionUtils.convertMap(customerMapper.selectList(CustomerDO::getId, customerIdList), CustomerDO::getId);
        for (ProjectSimpleListRespVO projectSimpleListRespVO : projectSimpleListRespVOList) {
            projectSimpleListRespVO.setManagerName(userMap.get(projectSimpleListRespVO.getManagerId()).getNickname());
            projectSimpleListRespVO.setDeptName(deptMap.get(projectSimpleListRespVO.getDeptId()).getName());
            CustomerDO customerDO = customerMap.get(projectSimpleListRespVO.getCustomerId());
            projectSimpleListRespVO.setCustomerName(customerDO == null ? "" : customerDO.getName());
        }
        return projectSimpleListRespVOList;
    }
