    @Override
    public PageResult<ManHourPageRespVO> getPage(ManHourPageReqVO reqVO) {
        Long userId = SecurityFrameworkUtils.getLoginUserId();
        List<Long> userRoleList = CollectionUtils.convertList(roleApi.getUserRole(userId).getData(), UserRoleDTO::getRoleId);
        if (!userRoleList.contains(FSTConstant.SUPER_ADMIN_ROLE)) {
            // 非超级管理员只能看到自己
            reqVO.setPersonId(userId);
        }
        if (StrUtils.isNotEmpty(reqVO.getPersonName())) {
            String personName = reqVO.getPersonName().trim();
            List<Long> personIds = userService.getUserMapByCache().values().stream()
                    .filter(user -> user.getNickname() != null && user.getNickname().contains(personName))
                    .map(UserInfoRespVO::getUserId)
                    .collect(Collectors.toList());
            if (CollectionUtils.isEmpty(personIds)) {
                return new PageResult<ManHourPageRespVO>().setList(Collections.emptyList()).setTotal(0);
            }
            reqVO.setPersonIds(personIds);
        }
        reqVO.setDataOrigin(FSTConstant.WorkHoursDataOrigin.PSA);

        PageResult<ManHourDO> pageResult = manHourMapper.selectPage(reqVO);
//        List<Long> manHourIdList = CollectionUtils.convertListSingle(pageResult.getList(), ManHourDO::getId);
//        List<CustomizedBpmOADO> customizedBpmOADOList = customizedBpmOAMapper.selectList(CustomizedBpmOADO::getParamsId, manHourIdList);
        // 流程实例ID集合 // TODO lfz：需要查询审批人
//        List<String> processInstanceIdList = CollectionUtils.convertListSingle(customizedBpmOADOList, CustomizedBpmOADO::getProcessInstanceId);

        return new PageResult<ManHourPageRespVO>().setList(this.convert(pageResult.getList())).setTotal(pageResult.getTotal());
    }
