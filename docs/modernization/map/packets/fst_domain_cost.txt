/**
 * OH 分摊确定性计算领域服务（纯 domain，无 Spring/无 Lombok，须在
 * {@code CostProjectCostBeanConfig} 显式 {@code @Bean} 装配）。
 *
 * <p>把待分摊 OH 池行按「档次（交付自有/服务/公共）× 性质（管理/销售）」两轴归档到项目工时基数：
 * 交付自有档按 {@code costOrgId} 归本部门 own 列、服务档按 {@code belongDeliveryId} 归归属交付部门 service 列、
 * 公共档累加主行。分母（工时基数）为 0 的档次顺延下期（记 {@link AllocationDraft.PendingReason}、不圈走、不上卷、不乱摊）。
 *
 * <p>本类分三个方法承接计划 Task 5/6/7：{@link #aggregate} 两轴聚合；后续 {@code derive} 派生占比四金额；
 * {@code verifyInvariants} 逐性质对账自校验。金额累加阶段保留原精度，四舍五入在派生阶段做。
 */
public final class OhAllocationCalculator {

    private static final int DEPT_ROLE_DELIVERY_OWN = 0;
    private static final int DEPT_ROLE_SERVICE = 1;
    private static final int DEPT_ROLE_PUBLIC = 2;

    private static final int EXPENSE_NATURE_MGMT = 0;

    /** 占比中间量精度（避免四舍五入丢占比）。 */
    private static final int RATIO_SCALE = 10;
    /** 币种金额精度（2 位）。 */
    private static final int AMOUNT_SCALE = 2;

    /**
     * 两轴聚合：把 {@code pooled} 按档次 × 性质归档到 {@code basis} 派生的工时基数分母，并判定分母为 0 顺延。
     */
    public AllocationAggregate aggregate(List<PooledOverhead> pooled, List<LedgerBasisRow> basis) {
        // 1. 工时基数分母：各交付部门 Σ notaxAmount（仅工时行），及全公司合计
        Map<Long, BigDecimal> basisByDept = new LinkedHashMap<>();
        BigDecimal companyBasisTotal = BigDecimal.ZERO;
        for (LedgerBasisRow row : basis) {
            if (!row.isTimesheetBasis()) {
                continue;
            }
            BigDecimal amount = row.getNotaxAmount();
            basisByDept.merge(row.getDeliveryDeptId(), amount, BigDecimal::add);
            companyBasisTotal = companyBasisTotal.add(amount);
        }

        // 2. 逐池行归档
        BigDecimal companyOhMgmt = BigDecimal.ZERO;
        BigDecimal companyOhSales = BigDecimal.ZERO;
        Map<Long, DeptAcc> deptAcc = new LinkedHashMap<>();
        Set<Long> pickedPoolLineIds = new LinkedHashSet<>();
        List<AllocationDraft.PendingReason> pendingReasons = new ArrayList<>();

        boolean companyBasisPositive = companyBasisTotal.compareTo(BigDecimal.ZERO) > 0;

        for (PooledOverhead oh : pooled) {
            if (oh.getDeptRole() == DEPT_ROLE_PUBLIC) {
                // 公共档：分母 = 全公司工时基数；为 0 则顺延
                if (!companyBasisPositive) {
                    pendingReasons.add(new AllocationDraft.PendingReason(oh.getCostOrgId(), oh.getNotaxAmount()));
                    continue;
                }
                if (oh.getExpenseNature() == EXPENSE_NATURE_MGMT) {
                    companyOhMgmt = companyOhMgmt.add(oh.getNotaxAmount());
                } else {
                    companyOhSales = companyOhSales.add(oh.getNotaxAmount());
                }
                pickedPoolLineIds.add(oh.getPoolLineId());
                continue;
            }

            // 部门档分母 dept：服务档归 belongDeliveryId；交付自有默认归 costOrgId，
            // 但组织重构后旧交付部门(承载 OH、自身无工时)会配置继任交付部门 belongDeliveryId(≠自己)，
            // 此时改挂继任交付部门的工时基数分摊（金额仍记 own 列，见 DeptAcc#add）——否则永远顺延（#1548）。
            Long targetDept;
            if (oh.getDeptRole() == DEPT_ROLE_SERVICE) {
                targetDept = oh.getBelongDeliveryId();
            } else {
                targetDept = oh.getBelongDeliveryId() != null
                        && !oh.getBelongDeliveryId().equals(oh.getCostOrgId())
                        ? oh.getBelongDeliveryId() : oh.getCostOrgId();
            }
            BigDecimal deptBasis = targetDept == null ? null : basisByDept.get(targetDept);
            if (deptBasis == null || deptBasis.compareTo(BigDecimal.ZERO) <= 0) {
                pendingReasons.add(new AllocationDraft.PendingReason(oh.getCostOrgId(), oh.getNotaxAmount()));
                continue;
            }
            DeptAcc acc = deptAcc.computeIfAbsent(targetDept, k -> new DeptAcc(deptBasis));
            acc.add(oh.getDeptRole(), oh.getExpenseNature(), oh.getNotaxAmount());
            pickedPoolLineIds.add(oh.getPoolLineId());
        }

        Map<Long, DeptBucket> deptBuckets = new LinkedHashMap<>();
        for (Map.Entry<Long, DeptAcc> e : deptAcc.entrySet()) {
            deptBuckets.put(e.getKey(), e.getValue().toBucket());
        }

        return new AllocationAggregate(companyOhMgmt, companyOhSales, companyBasisTotal,
                deptBuckets, basisByDept, companyBasisTotal, pickedPoolLineIds, pendingReasons);
    }

     * 占比 + 四金额派生 + 尾差归口：把两轴聚合结果摊到逐工时子行。
     *
     * <p>每条 {@code basis} 行产出一条子行（1:1）；非工时行两占比与四金额均 0 占位。工时行：
     * {@code deptRatio = notax / 部门工时基数}、{@code companyRatio = notax / 公司工时基数}（中间量高精度）；
     * 四金额 = 对应档次 × 性质 OH 小计 × 占比，{@code setScale(2, HALF_UP)}。尾差按（档次组 × 性质）分别求和后
     * 归该组内分摊额最大的工时子行，保证逐组闭合到 OH 小计。
     */
    public AllocationDraft derive(AllocationAggregate agg, List<LedgerBasisRow> basis) {
        Map<Long, BigDecimal> basisByDept = agg.getBasisByDept();
        Map<Long, DeptBucket> buckets = agg.getDeptBuckets();
        BigDecimal companyBaseTotal = agg.getCompanyBaseTotal();
        boolean companyBasisPositive = companyBaseTotal.compareTo(BigDecimal.ZERO) > 0;

        List<LineAcc> accs = buildAccs(basis);
        for (LineAcc acc : accs) {
            if (acc.timesheet) {
                fillTimesheetBase(acc, agg, basisByDept, buckets, companyBaseTotal, companyBasisPositive);
            }
        }
        // 尾差归口：公司档两性质（全部工时行）——归口行标记 companyTail
        reconcileCompany(accs, agg);
        // 尾差归口：部门档两性质（逐部门内工时行）——归口行标记 deptTail
        for (Map.Entry<Long, DeptBucket> e : buckets.entrySet()) {
            reconcileDept(accs, e.getKey(), e.getValue());
        }
        return assembleDraft(accs, buckets, agg);
    }
