/**
 * 结转行 → 凭证草稿组装端口实现（{@link PostingVoucherDrafter}）：回查成本台账 + 配对产值单，组三上下文
 * {@link PostingFact}（部门经 {@code psa_dept_u8c} 映射、凭证 9 收入组税腿），经 {@link PostingRuleProvider}
 * 取 {@code rule_category=3} 规则 → 调纯函数 {@link PostingRuleEngine} 渲一张平衡凭证（设计文档 §4.1/§4.2）。
 *
 * <p>token 名（{@code notax_amount}/{@code complete_amount}/{@code TBD_REVENUE_NOTAX}/{@code TBD_TAX}/
 * {@code cost_org_id}/{@code cuvalue.dept_id}）与 seed {@code V20260730144331} 精确对齐（凭证9客户核算维度已撤除，
 * 收入组只保留 {@code cuvalue.dept_id} 部门维度，见 #1111）。部门映射解析不出即不放入 fact，交引擎 {@code null_mode=0}
 * 抛 {@link PostingRuleException} 挡确认、绝不静默丢。
 */
@Component
public class PostingVoucherDrafterImpl implements PostingVoucherDrafter {

    /** 成本结转贷方动态科目占位 token（与 seed subject_code 及引擎 DYN- 约定逐字一致）：按 cost_type_code 解析 U8C 科目。 */
    private static final String COST_TYPE_ACCOUNT_TOKEN = "DYN-COST-TYPE-ACCT";

    /** 贷方费用腿金额 token · 普通费用（入账规则 DYN 行 amount_formula，与规则配置逐字一致）。 */
    private static final String EXPENSE_LEG_TOKEN = "expense_notax_amount";
    /** 贷方费用腿金额 token · 预付款（入账规则静态 A123 行 amount_formula，仅该行挂供应商辅助核算）。 */
    private static final String PREPAY_LEG_TOKEN = "prepay_notax_amount";

    /** 费用性质='0'（项目成本）：成本结转贷方动态科目只取这一档（psa_oa_account_u8c.fee_type，视图 varchar）。 */
    private static final String FEE_TYPE_PROJECT_COST = "0";

    /** 部门辅助核算维度码（红字头摘要 {@code {deliveryCenter}} 从聚合凭证的部门 aux 取值名，与引擎渲染同源）。 */
    private static final String AUX_DIM_DEPT = "部门";

    private final CostLedgerMapper costLedgerMapper;
    private final U8cDeptMapper u8cDeptMapper;
    private final CuValueMapper cuValueMapper;
    private final OaAccountU8cMapper oaAccountU8cMapper;
    private final PostingRuleProvider ruleProvider;
    private final PostingRuleEngine engine;

    private final CarryBatchRepository carryBatchRepository;

    public PostingVoucherDrafterImpl(CostLedgerMapper costLedgerMapper, U8cDeptMapper u8cDeptMapper,
                                     CuValueMapper cuValueMapper, OaAccountU8cMapper oaAccountU8cMapper,
                                     PostingRuleProvider ruleProvider,
                                     PostingRuleEngine engine, CarryBatchRepository carryBatchRepository) {
        this.costLedgerMapper = costLedgerMapper;
        this.u8cDeptMapper = u8cDeptMapper;
        this.cuValueMapper = cuValueMapper;
        this.oaAccountU8cMapper = oaAccountU8cMapper;
        this.ruleProvider = ruleProvider;
        this.engine = engine;
        this.carryBatchRepository = carryBatchRepository;
    }

    /**
     * 分块落账批量口径（#1195 M6）：与原单行落库口径同一组 fact + {@code engine.generate} 逐行独立成张（不聚合），
     * 但维表一趟 {@link #prefetch} + 规则一次加载，消除逐行 selectById/selectOne + 规则三表逐行重载。
     * 红冲不支持——阶段1 不制证（分块处理器已跳过），误入按编程错误 fail-fast。
     */
    @Override
    public Map<Long, VoucherDraft> draftPerLine(List<CarryBatchLine> lines, CarryPostingContext ctx) {
        if (lines == null || lines.isEmpty()) {
            return Collections.emptyMap();
        }
        if (CarryTypeEnum.REVENUE_RED_REVERSAL.getValue().equals(ctx.getCarryType())) {
            throw new IllegalStateException("红冲阶段1不做逐行制证（凭证由阶段2按退回单×部门聚合生成）");
        }
        FactSources sources = prefetch(lines);
        List<PostingRule> rules = ruleProvider.loadWipCarryRules();
        Map<Long, VoucherDraft> out = new LinkedHashMap<>(Math.max(16, lines.size() * 2));
        for (CarryBatchLine line : lines) {
            List<VoucherDraft> drafts = engine.generate(rules,
                    Collections.singletonList(buildFact(line, ctx, sources)));
            out.put(line.getId(), drafts.get(0));
        }
        return out;
    }

    /**
     * 产值红冲红字凭证（#1133 镜像取负重算）：对红冲行 {@code reversalOfLineId} 指向的<b>被冲原转出行</b>，按当前记账规则
     * 用 {@link #buildFact}（{@code carryType=WIP_WRITE_OFF}）重算原转出正向分录，走正向 {@link PostingRuleEngine#generateAggregated}
     * 按 公司×凭证字×部门 聚合，最后<b>整张取负</b> → 负向凭证 9。聚合边界收窄到<b>单张产值退回单</b>：以红冲行的
     * {@code pairedCuValueId}（退回行 id）分组，组内按部门聚合、不同退回单不合并（保留"哪张退回单→哪张红字凭证"可追溯）。
     *
     * <p>N+1：原行一趟 {@link CarryBatchRepository#findLinesByIds} IN 载入、维表一趟 {@link #prefetch}，分组/聚合全在内存。
     * 头摘要在 {@link #negateWhole} 层单独拼（带被退回产值单号），不污染引擎共享摘要模板。经引擎后可能抛
     * fail-closed {@link PostingRuleException}（缺部门映射/费用类别科目），同正向包成财务可读异常。
     */
    private List<VoucherDraft> redReversalVouchers(List<CarryBatchLine> redLines, CarryPostingContext ctx) {
        // 1. 批量载被冲原转出行（一趟 IN 消 N+1），按 id 建 map
        Set<Long> originIds = new LinkedHashSet<>();
        for (CarryBatchLine rl : redLines) {
            if (rl.getReversalOfLineId() != null) {
                originIds.add(rl.getReversalOfLineId());
            }
        }
        Map<Long, CarryBatchLine> originById = new HashMap<>();
        for (CarryBatchLine o : carryBatchRepository.findLinesByIds(originIds)) {
            originById.put(o.getId(), o);
        }
        // 红冲行 → 对应原转出行（缺失即防御性抛，正常由圈选 screenAll 前置拦下 ORIGIN_NOT_FOUND）
        List<CarryBatchLine> originLines = new ArrayList<>(redLines.size());
        for (CarryBatchLine rl : redLines) {
            CarryBatchLine origin = originById.get(rl.getReversalOfLineId());
            if (origin == null) {
                throw new PostingRuleException("红冲找不到被冲原转出行 reversalOfLineId=" + rl.getReversalOfLineId());
            }
            originLines.add(origin);
        }
        // 2. 预取原行维表（喂 origin 行，复用正向 prefetch，零改）
        FactSources src = prefetch(originLines);
        // carryType 用 WIP_WRITE_OFF：让 buildFact 塞进 fact 的 carryType 命中正向转出规则（红冲 type3 不匹配任何正向规则）
        CarryPostingContext forwardCtx = new CarryPostingContext(ctx.getBatchId(), ctx.getBatchNo(),
                ctx.getPeriodId(), ctx.getPeriodName(), ctx.getCompanyCode(),
                CarryTypeEnum.WIP_WRITE_OFF.getValue(), ctx.getOperatorId());
        List<PostingRule> rules = ruleProvider.loadWipCarryRules();
        // 3. 按退回单分组（红冲行 pairedCuValueId=退回行 id），组内按部门聚合、不同退回单不合并
        Map<Long, List<CarryBatchLine>> byReturnSlip = new LinkedHashMap<>();
        for (CarryBatchLine rl : redLines) {
            byReturnSlip.computeIfAbsent(rl.getPairedCuValueId(), k -> new ArrayList<>()).add(rl);
        }
        List<VoucherDraft> out = new ArrayList<>();
        try {
            for (Map.Entry<Long, List<CarryBatchLine>> group : byReturnSlip.entrySet()) {
                Long returnSlipId = group.getKey();   // 退回单 id（=红冲行 pairedCuValueId），作 external_no 唯一位
                List<CarryBatchLine> groupRedLines = group.getValue();
                List<PostingFact> facts = new ArrayList<>(groupRedLines.size());
                String cuValueCode = null;   // 同组恒同 = 被退回产值单号（取组内首个非空 origin.paired_cu_value_code）
                for (CarryBatchLine rl : groupRedLines) {
                    CarryBatchLine origin = originById.get(rl.getReversalOfLineId());
                    facts.add(buildFact(origin, forwardCtx, src));
                    if (cuValueCode == null && origin.getPairedCuValueCode() != null) {
                        cuValueCode = origin.getPairedCuValueCode();
                    }
                }
                // 复用正向聚合（公司×凭证字×部门 每部门一张、整张平衡），逐张整张取负 → 负向凭证 9；
                // 带 returnSlipId 让 external_no 走「退回单×部门」唯一编号（避同部门多退回单撞键，#1133）
                for (VoucherDraft fwd : engine.generateAggregated(rules, facts)) {
                    out.add(negateWhole(fwd, redSummary(ctx.getPeriodName(), deliveryCenterOf(fwd), cuValueCode),
                            returnSlipId));
                }
            }
        } catch (PostingRuleException e) {
            // 经引擎后可能抛 fail-closed（缺部门 U8C 映射 / 费用类别缺科目）：同正向包成财务可读提示（入参用 origin 行 + src）
            String deptFriendly = describeUnmappedCostOrgs(originLines, src);
            if (deptFriendly != null) {
                throw new CarryDeptUnmappedException(deptFriendly);
            }
            String costTypeFriendly = describeUnmappedCostTypes(originLines, src);
            if (costTypeFriendly != null) {
                throw new CarryCostTypeUnmappedException(costTypeFriendly);
