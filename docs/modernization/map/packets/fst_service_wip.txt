    @Override
    public WIPCarryForwardRespVO getWIPCarriedOverList(WIPCarryForwardPageReqVO reqVO) {
        // 产值确认时间区间
        String[] completeDates = DateUtil.getStartAndEndByMonth(
                reqVO.getYear() + "-" + StrUtils.autoGenericCode(reqVO.getMonth().toString(), 2) + "-01");
        reqVO.setCompleteDate(completeDates);
        // 如果选择时间大于现在，返回空
        if (new DateTime(completeDates[0]).isAfterNow())
            return new WIPCarryForwardRespVO().setDataList(new ArrayList<>());

        // 主表
        List<CarryForwardDataDO> carryForwardDataList = carryForwardDataMapper.selectList(
                new LambdaQueryWrapperX<CarryForwardDataDO>()
                        .eqIfPresent(CarryForwardDataDO::getWriteOffFlag, FSTConstant.IsOrNo.是)
                        .eqIfPresent(CarryForwardDataDO::getCarryForwardType, FSTConstant.CarryForwardType.WIP)
                        .eqIfPresent(CarryForwardDataDO::getYear, reqVO.getYear())
                        .eqIfPresent(CarryForwardDataDO::getMonth, reqVO.getMonth()));
        if (carryForwardDataList.size() < 1)
            return new WIPCarryForwardRespVO().setDataList(new ArrayList<>());
        Long pId = carryForwardDataList.get(0).getId();
        reqVO.setCarryForwardDataId(pId);
        // 子表
        List<CarryForwardDataWorksDO> worksDOList = carryForwardDataWorksMapper.selectList(
                new LambdaQueryWrapperX<CarryForwardDataWorksDO>()
                        .eqIfPresent(CarryForwardDataWorksDO::getWipId, pId)
                        .eqIfPresent(CarryForwardDataWorksDO::getWriteOffFlag, reqVO.getWriteOffFlag()));

        // 0：直接人工成本；1：日常费用；2：差旅费用；3：付款申请；4：特殊付款；5：委外；6：信息服务
        Map<String, List<BigDecimal>> termNumberPayAmountMap = new HashMap<>();
        for (CarryForwardDataWorksDO carryForwardDataWorksDO : worksDOList) {
            List<BigDecimal> amountList = termNumberPayAmountMap.get(carryForwardDataWorksDO.getTermNumber());
            if (CollectionUtils.isEmpty(amountList)) {
                amountList = Arrays.asList(BigDecimal.ZERO, BigDecimal.ZERO, BigDecimal.ZERO, BigDecimal.ZERO,
                        BigDecimal.ZERO, BigDecimal.ZERO, BigDecimal.ZERO);
                termNumberPayAmountMap.put(carryForwardDataWorksDO.getTermNumber(), amountList);
            }
            Integer index = null;
//            CarryForwardDataTypeEnum.describeOfValue()
            if (FSTConstant.CarryForwardDataType.工时.equals(carryForwardDataWorksDO.getType()))
                index = 0;
            if (FSTConstant.CarryForwardDataType.日常费用.equals(carryForwardDataWorksDO.getType()))
                index = 1;
            if (FSTConstant.CarryForwardDataType.差旅费用.equals(carryForwardDataWorksDO.getType()))
                index = 2;
            if (FSTConstant.CarryForwardDataType.付款申请.equals(carryForwardDataWorksDO.getType()))
                index = 3;
            if (FSTConstant.CarryForwardDataType.特殊付款.equals(carryForwardDataWorksDO.getType()))
                index = 4;
            if (FSTConstant.CarryForwardDataType.委外.equals(carryForwardDataWorksDO.getType()))
                index = 5;
            if (FSTConstant.CarryForwardDataType.信息服务.equals(carryForwardDataWorksDO.getType()))
                index = 6;
            if (index == null)
                continue;

            BigDecimal amount = amountList.get(index);
            amount = amount.add(carryForwardDataWorksDO.getAmount() == null ? BigDecimal.ZERO : carryForwardDataWorksDO.getAmount());
            amountList.set(index, amount);
        }
