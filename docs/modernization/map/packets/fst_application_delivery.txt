/**
 * 命令用例：任务包执行期操作。装载（带项目归属校验）→ 聚合业务方法 → 增量落库。
 * <p>预算基线守卫（refs #1005）：审批中任务包结构冻结（增/删/改一律拒）；锁定后预算工时冻结
 * （调整走预算变更）。执行状态流转（完成/撤销/返工/ETC 审批）不属于基线，不受两闸影响。
 * <p><b>ETC 审批发起三步（refs #1811）</b>：冻结态 + 发起轮次先在事务 A 提交 → 事务外调 bpm →
 * 事务 B 绑定实例号。bpm 可能在发起调用内就跑完流程并投递终态消息（单节点自动通过），拆开后终态消息
 * 到达时冻结态必已可见，消费者直接落终态；bpm 失败由补偿事务还原冻结态。事务边界用 TransactionTemplate
 * 显式控制（同类内自调用不经代理，注解式失效）。
 */

    @Override
    public void markDone(Long projectId, Long taskId) {
        this.mutateByTask(projectId, taskId, progress -> progress.markTaskDone(taskId));
    }

    @Override
    public void markDoneBySelf(Long projectId, Long taskId) {
        // 指派人闸（refs #1971）：本入口不挂进度管理权限点，登录态即可调用，
        // 必须在用例内锁死「只能完成指派给自己的包」，其余流转规则与 PM 的完成完全同源
        Long operatorId = cn.iocoder.yudao.framework.security.core.util.SecurityFrameworkUtils.getLoginUserId();
        this.mutateByTask(projectId, taskId, progress -> {
            TaskPackage target = progress.getTasks().stream()
                    .filter(t -> taskId.equals(t.getId()))
                    .findFirst().orElse(null);
            if (target == null || !java.util.Objects.equals(target.getAssigneeId(), operatorId)) {
                throw new ServiceException(ErrorCodeConstants.PROGRESS_TASK_DONE_NOT_ASSIGNEE);
            }
            return progress.markTaskDone(taskId);
        });
    }

    @Override
    public void reopen(Long projectId, Long taskId) {
        this.mutateByTask(projectId, taskId, progress -> progress.reopenTask(taskId));
    }

    @Override
    public void rework(Long projectId, Long taskId) {
        DeliverableProgress progress = deliverableProgressRepository.findByTaskId(projectId, taskId);
        if (progress == null) {
            throw new ServiceException(ErrorCodeConstants.PROGRESS_TASK_NOT_EXISTS);
        }
        // 成果闸（refs #847/#1887）：已驳回（DLV_DRAFT）不占名额、此处查不到，原地返工放行；
        // 审批中一律拒——审批依据不可在审批期变动，发起人须先去审批中心结束流程；
        // 已审批通过的成果是既成事实（收入 / 产值已固化），不能原地返工，走 reworkByNewDeliverable 新建返工包
        cn.iocoder.yudao.module.fst.domain.delivery.project.DeliverableSubmission occupying =
                submissionRepository.findOccupyingByDeliverableId(progress.getId());
        if (occupying != null) {
            throw new ServiceException(
                    cn.iocoder.yudao.module.fst.domain.delivery.project.DeliverableSubmissionStatus
                            .isApproved(occupying.getStatus())
                            ? ErrorCodeConstants.DELIVERABLE_REWORK_APPROVED_USE_NEW_DELIVERABLE
                            : ErrorCodeConstants.DELIVERABLE_REWORK_LOCKED_BY_SUBMISSION);
        }
        TaskPackage task;
        try {
            task = progress.reworkTask(taskId);
        } catch (ProgressRuleViolation violation) {
            throw ProgressErrors.translate(violation);
        }
        this.persistStatusOrThrow(task);
    }

    @Override
    public boolean setEtc(ProgressTaskCommand command) {
        // refs #818/#1265：需审批的 ETC 调增走 PM_ETC，原 ETC 保持生效，终态回写用例落地；
        // 调低/持平直接生效落库（同一写点：pending/process 本就为空，复核标记一并清）。
        // 触发口径：调增后项目预测人工成本总额 > 预算人工成本总额才审批，仍在预算内的调增直接生效。
        // refs #1811：发起拆三步（冻结态先提交 → 事务外调 bpm → 绑实例号），不再整体一个事务
        DeliverableProgress progress = deliverableProgressRepository
                .findByTaskId(command.getProjectId(), command.getTaskId());
        if (progress == null) {
            throw new ServiceException(ErrorCodeConstants.PROGRESS_TASK_NOT_EXISTS);
        }
        // 退回发起人重提（refs #1568）：审批中且流程实例停在发起人节点——覆盖待批值/原因后
        // 重新发起，bpm 识别同一存活实例 → 覆盖变量 + 完成发起人任务，同实例续跑
        boolean etcPendingNow = progress.getTasks().stream()
                .anyMatch(item -> command.getTaskId().equals(item.getId()) && item.isEtcPending());
        if (etcPendingNow && workflowReturnStateQuery.isReturnedToInitiator(command.getTaskId())) {
            return Boolean.TRUE.equals(this.tx().execute(status -> this.resubmitReturnedEtc(command, progress)));
        }
        DeliverableProgress.EtcChange change;
        java.math.BigDecimal oldEtcHours;
        try {
            oldEtcHours = progress.getTasks().stream()
                    .filter(item -> command.getTaskId().equals(item.getId()))
                    .findFirst().map(TaskPackage::getEtcHours).orElse(null);
            change = progress.submitEtcChange(command.getTaskId(), command.getEtcHours(), command.getEtcReason(),
                    this.laborCostHeadroom(command.getProjectId(), command.getTaskId()));
        } catch (ProgressRuleViolation violation) {
            throw ProgressErrors.translate(violation);
        }
        if (!change.isApprovalRequired()) {
            // 调低/持平：单条 CAS 直改；库中已是 PENDING（他人抢先发起）即让路，不覆盖冻结态（refs #1286）
            if (!deliverableProgressRepository.updateEtcApproval(change.getTask(), false)) {
                throw new ServiceException(ErrorCodeConstants.PROGRESS_TASK_ETC_APPROVAL_PENDING);
            }
            return false;
        }
        // 事务 A：冻结态 + 发起轮次一并提交（CAS 单飞闸：库中已 PENDING 即拒，不发起第二个流程）
        WorkflowStartRoundDO round = this.tx().execute(status -> this.freezeAndBeginRound(change.getTask()));
        this.startAndBindEtcApproval(command.getProjectId(), progress, change.getTask(), oldEtcHours, round, null);
        return true;
    }

    /**
     * 项目人工成本余量：从进度树读模型取横幅同款的预算人工成本 / 预测人工成本，以及本任务包的执行费率。
     * <p>读不到进度树或任务包节点（理论上不会发生：聚合已按 projectId + taskId 装载成功）时返回 null，
     * 聚合按「调高即审批」的保守口径处理——宁可多审一次，也不在判据缺失时放行。
     */
    private cn.iocoder.yudao.module.fst.domain.delivery.project.LaborCostHeadroom laborCostHeadroom(
            Long projectId, Long taskId) {
        return cn.iocoder.yudao.module.fst.domain.delivery.project.LaborCostHeadroom
                .fromTree(progressTreeQuery.byProjectId(projectId), taskId);
    }
