// 结转后 WIP 阶段：按（合同状态 + POC）自动判定（仅判定阶段、不折算金额）
function _wipStage(contractStatus, poc) {
    if (contractStatus === 'NOT_RETURNED') return 'PRE_CONTRACT'
    return poc >= 100 ? 'DONE' : 'IN_PROGRESS'
}

// 阶段前驱：阶段递进(carry_type=1)的"源状态"=成本结转产出阶段的上一阶段
function _prevStage(stage) {
    if (stage === 'DONE') return 'IN_PROGRESS'
    if (stage === 'IN_PROGRESS') return 'PRE_CONTRACT'
    return 'PRE_CONTRACT'
}
