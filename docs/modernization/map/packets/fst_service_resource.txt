    @Override
    public PageResult<ResourcePageItemRespVO> getTaskPage(ResourcePageReqVO reqVO) {
        Map<Long, DeptRespDTO> deptMap = CollectionUtils.convertMap(deptApi.getDeptsByCache().getData(), DeptRespDTO::getId);
        Map<Long, PostRespDTO> postMap = CollectionUtils.convertMap(postApi.getPostsByCache().getData(), PostRespDTO::getId);
        AdminUserRespDTO dto = new AdminUserRespDTO();
        dto.setDeptIds(reqVO.getDeptIds());
        dto.setNickname(reqVO.getUserName());
        dto.setBilling(FSTConstant.BillingUser.是);
        dto.setId(reqVO.getUserId());
        // 条件查询所有人员
        CommonResult<List<AdminUserRespDTO>> userResult = adminUserApi.getUsersByCondition(dto);
        List<AdminUserRespDTO> userList = userResult.getData();

        List<Long> userIdList = CollectionUtils.convertList(userList, AdminUserRespDTO::getId);

        reqVO.setUserIds(userIdList);
        String beginDate = reqVO.getBeginDate();
        if (beginDate == null) {
            // 默认从今天开始
            reqVO.setBeginDate(new DateTime().toString("yyyy-MM-dd"));
        }
        // 条件查询所有任务
        List<TaskDO> taskDOList = taskMapper.selectByCondition(reqVO);
        // 计算闲忙度开始 ===
        CalendarReqVO calendarReqVO = new CalendarReqVO();
        calendarReqVO.setWorkday(1);
        calendarReqVO.setDatetime(new String[]{reqVO.getBeginDate(), reqVO.getEndDate()});
        // 工作日天数
        Long count = calendarMapper.selectCountByCondition(calendarReqVO);
        // 时间段内的理应工时
        BigDecimal workHours = new BigDecimal(count).multiply(new BigDecimal(CommonConstants.日小时));
        List<ResourcePageItemRespVO> resultDataList = new ArrayList<>();
        long tempId = 1L;
        for (AdminUserRespDTO user : userList) {
            Long userId = user.getId();
            user.setDeptName(deptMap.get(user.getDeptId()).getName());
            // 每个人总的任务小时数
            BigDecimal taskHours = BigDecimal.ZERO;
            List<ResourcePageItemRespVO> itemList = new ArrayList<>();

            ResourcePageItemRespVO resourcePageItemRespVO = new ResourcePageItemRespVO();
            resourcePageItemRespVO.setUserId(user.getId());
            resourcePageItemRespVO.setDeptId(user.getDeptId());
            resourcePageItemRespVO.setUserName(user.getNickname());
            resourcePageItemRespVO.setDeptName(deptMap.get(user.getDeptId()).getName());
            resourcePageItemRespVO.setPostName((user.getPostIds() != null && user.getPostIds().size() > 0)
                    ? postMap.get(new ArrayList<>(user.getPostIds()).get(0)).getName()
                    : "暂无岗位");

            resourcePageItemRespVO.setName("--");
            resourcePageItemRespVO.setTheoreticalWorkHours(workHours);
            resourcePageItemRespVO.setAssignHours(taskHours);
            resourcePageItemRespVO.setPersonId(user.getId());
            resourcePageItemRespVO.setPersonName(user.getNickname());
            resourcePageItemRespVO.setId(tempId);

            for (TaskDO taskDO : taskDOList) {
                if (userId.equals(taskDO.getPersonId())) {
                    ResourcePageItemRespVO respVO = new ResourcePageItemRespVO();
                    respVO.setUserId(user.getId());
                    respVO.setUserName(user.getNickname());
                    respVO.setDeptId(user.getDeptId());
                    respVO.setDeptName(deptMap.get(user.getDeptId()).getName());
                    Set<Long> postIds = user.getPostIds();

                    respVO.setPostName(
                            (postIds != null && postIds.size() > 0)
                                    ? postMap.get(new ArrayList<>(postIds).get(0)).getName()
                                    : "暂无岗位");
                    respVO.setName(taskDO.getName());
                    taskHours = taskHours.add(taskDO.getAssignHours());
                    respVO.setPlanStartDate(taskDO.getPlanStartDate());
                    respVO.setPlanEndDate(taskDO.getPlanEndDate());
                    respVO.setPersonName(user.getNickname());
                    respVO.setParentId(tempId);
                    respVO.setFlagKeyTask(taskDO.getFlagKeyTask());
                    itemList.add(respVO);
                }
            }
            tempId = tempId + 1L;
            BigDecimal busyness = BigDecimalUtil.divide(taskHours, workHours, RoundingMode.HALF_UP);
            resourcePageItemRespVO.setBusyness(busyness);
            // 获取闲忙度条件
            String busynessStr = reqVO.getBusyness();
            if (!StringUtils.isEmpty(busynessStr)) {
                String[] range = busynessStr.split(",");
                // 符合条件
                if (busyness.compareTo(new BigDecimal(range[0])) >= 0 && busyness.compareTo(new BigDecimal(range[1])) < 0) {
                    resultDataList.add(resourcePageItemRespVO);
                    resultDataList.addAll(itemList);
                }
            } else { // 没有条件
                resultDataList.add(resourcePageItemRespVO);
                resultDataList.addAll(itemList);
            }
        }
        return PageUtils.getPageResult(resultDataList, reqVO.getPageNo(), reqVO.getPageSize());
    }
