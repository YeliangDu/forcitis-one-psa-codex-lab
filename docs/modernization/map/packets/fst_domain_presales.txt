/**
 * 报价单聚合根（psa_quotation + 明细/成本树/用印三张子表，Q1~Q8）。
 * <p>不变式与状态机（终态口径见 {@link QuotationStatus}）：
 * <ul>
 *   <li>仅 DRAFT/REJECTED 可编辑与删除（{@link #ensureEditable()}）</li>
 *   <li>提交前置：名称必填、最终报价 &gt; 0；提交置 PENDING 并清空上一轮驳回意见</li>
 *   <li>仅 PENDING 可决策；驳回必填意见</li>
 *   <li>三汇总快照后端重算（{@link #applyEditorChanges}）：最终报价=明细合计+尾差；
 *       成本=Σ工时×成本费率+其他费用；毛利率按不含税口径 4 位小数</li>
 * </ul>
 * 无公共 setter；违例抛 {@link QuotationRuleViolation}，由应用层翻译为带错误码的 ServiceException（B12）。
 * <p>状态枚举复用 {@code fst-api} 的 {@link QuotationStatus}（枚举 SSOT，X1），不在领域层另建副本。
 */
public class Quotation {

    /** 默认整单税率（DDL 默认 0.06，毛利率不含税换算用） */
    private static final BigDecimal DEFAULT_TAX_RATE = new BigDecimal("0.06");

    /** 报价单ID（16位雪花，新建为 null 由仓储落库后绑定） */
    private Long id;
    /** 报价编号（QUO- 前缀后端生成，唯一） */
    private final String code;
    /** 所属商机ID */
    private final Long opportunityId;
    /** 转合同后回填的合同编号（只读展示） */
    private final String contractCode;
    /** 客户ID（发起时自商机回填快照） */
    private final Long customerId;
    /** 申请人ID */
    private final Long applyUserId;
    /** 申请人所属部门快照（组织调整不影响历史单据口径） */
    private final Long applyDeptId;
    /** 项目交付部门ID（发起时自商机回填快照） */
    private final Long deliverDeptId;
    /** 商机所属区域部门ID（发起时自商机 pdc_dept_id 快照，refs #1161；存量单为 null 读商机现值） */
    private Long pdcDeptId;
    /** 销售负责人快照（发起时自商机 manager_id 定格，refs #1173；NULL=存量单读商机现值） */
    private Long salesUserId;
    /** 整单税率 */
    private BigDecimal taxRate;
    /** 最终报价（明细合计+尾差，保存时重算快照） */
    private BigDecimal finalAmount;
    /** BU 预估成本（人力+其他费用，保存时重算快照） */
    private BigDecimal buCost;
    /** 人力成本快照（Σ工作包 工时×成本费率，保存/审批采纳树时聚合重算，refs #1173） */
    private BigDecimal laborCost;
    /** 委外及其他费用（固定 8 类目主表列直存，refs #1173） */
    private OtherCosts otherCosts;
    /** 毛利率（不含税口径 4 位小数；报价额≤0 时为 null） */
    private BigDecimal grossMargin;
    /** 销售团队可见范围（用户 id 字符串数组，JSON 落 visible_team） */
    private List<String> visibleTeam;
    /** 状态（quotation_status：DRAFT/PENDING/APPROVED/REJECTED） */
    private String status;
    /** 报价单名称（提交必填） */
    private String quoteName;
    /** 是否需要用印 */
    private Boolean needSeal;
    /** 尾差调整（可负） */
    private BigDecimal tailAdjust;
    /** 报价说明 */
    private String quoteNote;
    /** 低毛利原因说明 */
    private String lowGmReason;
    /** OA 驳回意见（重新提交时清空） */
    private String rejectReason;
    /** 提交 OA 时间 */
    private LocalDateTime submitTime;
    /** BPM 流程实例标识（OPP_QUOTATION；提交发起后由终态消息补写，null=未绑定/在途未知） */
    private String processInstanceId;
    /** 退回发起人放行标记（refs #1568，内存态不落库）：见 {@link #markReturnedToInitiator()} */
    private boolean returnedToInitiator;
    /** 乐观锁版本（草稿创建置 0） */
    private final Long version;
    /** 创建时间（表列 NOT NULL） */
    private final LocalDateTime createdTime;

    /** 报价单附件（编辑保存时整批替换，JSON 落 quote_file_ids，refs #1042） */
    private List<QuotationFile> quoteFiles = Collections.emptyList();
    /** 明细行（编辑保存时整批替换） */
    private List<QuotationLine> lines = Collections.emptyList();
    /** 成本测算树（编辑保存时整棵替换） */
    private List<CostEstimateNode> costNodes = Collections.emptyList();
    /** 成本测算业务分类（refs #1119）：随套用模板落定，空=未自模板起步的存量单 */
    private String costCategory;
    /** 城市分类编码（refs #1146）：岗位费率按 岗位+城市 取数的城市来源；空=存量单，费率不过滤城市 */
    private String cityCategoryCode;
    /** 用印登记（编辑保存时整批替换） */
    private List<QuotationSeal> seals = Collections.emptyList();

    private Quotation(Long id, String code, Long opportunityId, String contractCode, Long customerId,
                      Long applyUserId, Long applyDeptId, Long deliverDeptId, BigDecimal taxRate,
                      BigDecimal finalAmount, BigDecimal buCost, BigDecimal grossMargin, List<String> visibleTeam,
                      String status, String quoteName, Boolean needSeal, BigDecimal tailAdjust, String quoteNote,
                      String lowGmReason, String rejectReason, LocalDateTime submitTime, String processInstanceId,
                      Long version, LocalDateTime createdTime) {
        this.id = id;
        this.code = code;
        this.opportunityId = opportunityId;
        this.contractCode = contractCode;
        this.customerId = customerId;
        this.applyUserId = applyUserId;
        this.applyDeptId = applyDeptId;
        this.deliverDeptId = deliverDeptId;
        this.taxRate = taxRate;
        this.finalAmount = finalAmount;
        this.buCost = buCost;
        this.grossMargin = grossMargin;
        this.visibleTeam = visibleTeam == null ? Collections.emptyList() : visibleTeam;
        this.status = status;
        this.quoteName = quoteName;
        this.needSeal = needSeal;
        this.tailAdjust = tailAdjust;
        this.quoteNote = quoteNote;
        this.lowGmReason = lowGmReason;
        this.rejectReason = rejectReason;
        this.submitTime = submitTime;
        this.processInstanceId = processInstanceId;
        this.version = version;
        this.createdTime = createdTime;
    }

    /**
     * 发起报价草稿（F3）：编号由 {@link QuotationCodeGenerator} 按流水号规则生成后传入（refs #595），
     * 自商机回填客户/交付部门，默认税率，状态 DRAFT。
     */
    public static Quotation createDraft(String code, Long opportunityId, Long customerId, Long deliverDeptId,
                                        Long pdcDeptId, Long salesUserId, Long applyUserId, Long applyDeptId) {
        Quotation draft = new Quotation(null, code, opportunityId, null,
                customerId, applyUserId, applyDeptId, deliverDeptId, DEFAULT_TAX_RATE,
                null, null, null, null, QuotationStatus.DRAFT.getValue(),
                null, null, null, null, null, null, null, null, 0L, LocalDateTime.now());
        // 商机所属区域/销售负责人快照（refs #1161/#1173）：发起时刻定格，历史单据口径不随商机变更漂移
        draft.pdcDeptId = pdcDeptId;
        draft.salesUserId = salesUserId;
        return draft;
    }

     * 仓储回装（从持久化态重建聚合主单，不触发业务校验；子表按需另行装载）。

    public static Quotation restore(Long id, String code, Long opportunityId, String contractCode, Long customerId,

                                    List<String> visibleTeam, String status, String quoteName, Boolean needSeal,
